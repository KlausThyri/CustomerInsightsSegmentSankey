#if FABRIC_API
using System.Net;
using System.Text;
using Azure.Core;
using CustomerInsightsSegmentSankey.FabricApi;
using Xunit;

namespace CustomerInsightsSegmentSankey.FabricApi.Tests;

public sealed class FabricCapacityServiceTests
{
    private const string ResourceId =
        "/subscriptions/11111111-1111-1111-1111-111111111111/resourceGroups/fabric-rg/providers/Microsoft.Fabric/capacities/testcapacity";

    [Fact]
    public async Task EnsureActiveAsync_DoesNotResumeActiveCapacity()
    {
        var handler = new SequenceHandler(Response(HttpStatusCode.OK, "Active"));
        var service = CreateService(handler);

        await service.EnsureActiveAsync(CancellationToken.None);

        Assert.Single(handler.Requests);
        Assert.Equal(HttpMethod.Get, handler.Requests[0].Method);
        Assert.Equal(
            "https://management.azure.com" + ResourceId + "?api-version=2023-11-01",
            handler.Requests[0].Uri);
    }

    [Fact]
    public async Task EnsureActiveAsync_ResumesPausedCapacityAndWaitsForActiveState()
    {
        var handler = new SequenceHandler(
            Response(HttpStatusCode.OK, "Paused"),
            Response(HttpStatusCode.OK, "Paused"),
            new HttpResponseMessage(HttpStatusCode.Accepted),
            Response(HttpStatusCode.OK, "Resuming"),
            Response(HttpStatusCode.OK, "Active"));
        var service = CreateService(handler);

        await service.EnsureActiveAsync(CancellationToken.None);

        Assert.Equal(HttpMethod.Post, handler.Requests[2].Method);
        Assert.EndsWith(
            "/resume?api-version=2023-11-01",
            handler.Requests[2].Uri,
            StringComparison.Ordinal);
    }

    [Fact]
    public async Task EnsureActiveAsync_ReturnsRetryableStartingErrorWhileResumeContinues()
    {
        var responses = new List<HttpResponseMessage>
        {
            Response(HttpStatusCode.OK, "Paused"),
            Response(HttpStatusCode.OK, "Paused"),
            new(HttpStatusCode.Accepted)
        };
        responses.AddRange(Enumerable.Range(0, 6).Select(_ => Response(HttpStatusCode.OK, "Resuming")));
        var service = CreateService(new SequenceHandler(responses.ToArray()));

        var error = await Assert.ThrowsAsync<FabricQueryException>(
            () => service.EnsureActiveAsync(CancellationToken.None));

        Assert.Equal("capacity_starting", error.Code);
        Assert.Contains("starting automatically", error.Message, StringComparison.OrdinalIgnoreCase);
    }

    [Fact]
    public async Task EnsureActiveAsync_ExplainsMissingManagedIdentityPermission()
    {
        var service = CreateService(
            new SequenceHandler(new HttpResponseMessage(HttpStatusCode.Forbidden)));

        var error = await Assert.ThrowsAsync<FabricQueryException>(
            () => service.EnsureActiveAsync(CancellationToken.None));

        Assert.Equal("capacity_permission_missing", error.Code);
        Assert.Contains("Setup Center", error.Message, StringComparison.Ordinal);
    }

    private static FabricCapacityService CreateService(SequenceHandler handler) =>
        new(
            new HttpClient(handler),
            new StaticTokenCredential(),
            ResourceId,
            static (_, _) => Task.CompletedTask);

    private static HttpResponseMessage Response(HttpStatusCode status, string state) =>
        new(status)
        {
            Content = new StringContent(
                "{\"properties\":{\"state\":\"" + state + "\"}}",
                Encoding.UTF8,
                "application/json")
        };

    private sealed class StaticTokenCredential : TokenCredential
    {
        public override AccessToken GetToken(
            TokenRequestContext requestContext,
            CancellationToken cancellationToken) =>
            new("token", DateTimeOffset.MaxValue);

        public override ValueTask<AccessToken> GetTokenAsync(
            TokenRequestContext requestContext,
            CancellationToken cancellationToken) =>
            ValueTask.FromResult(new AccessToken("token", DateTimeOffset.MaxValue));
    }

    private sealed class SequenceHandler(params HttpResponseMessage[] responses)
        : HttpMessageHandler
    {
        private readonly Queue<HttpResponseMessage> responses = new(responses);

        public List<(HttpMethod Method, string Uri)> Requests { get; } = [];

        protected override Task<HttpResponseMessage> SendAsync(
            HttpRequestMessage request,
            CancellationToken cancellationToken)
        {
            Requests.Add((request.Method, request.RequestUri!.ToString()));
            return Task.FromResult(responses.Dequeue());
        }
    }
}
#endif
