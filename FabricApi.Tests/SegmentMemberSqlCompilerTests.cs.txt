    [Fact]
    public void CompileMemberQuery_Stage_SelectsStageCteDirectlyAndProjectsSortColumns()
    {
        var result = Compile(
            TwoStageQuery(),
            Options(stageOrder: 1, viewMode: "stage"),
            Table(
                "dataverse",
                "contact",
                "contactid",
                "statecode",
                "address1_country",
                "fullname"));

        Assert.DoesNotContain("EXCEPT", result.Sql);
        Assert.Contains("] AS m", result.Sql);
        Assert.Contains(
            "INNER JOIN [dataverse].[contact] AS p",
            result.Sql);
        Assert.Contains(
            "ON TRY_CONVERT(uniqueidentifier, p.[contactid]) = m.ProfileId",
            result.Sql);
        Assert.Contains("SortIsNull", result.Sql);
        Assert.Contains("SortValue", result.Sql);
        Assert.Contains("ORDER BY root.SortIsNull ASC, root.SortValue ASC, root.ProfileId ASC;", result.Sql);
        Assert.Equal("contact", result.ProfileEntity);
        Assert.Contains(11, result.Parameters.Select(p => p.Value));
    }

    [Fact]
    public void CompileMemberQuery_Removed_UsesPreviousExceptSelected()
    {
        var result = Compile(
            TwoStageQuery(),
            Options(stageOrder: 1, viewMode: "removed"),
            Table("dataverse", "contact", "contactid", "statecode", "address1_country", "fullname"));

    [Fact]
    public void CompileMemberQuery_ReusesSameStageCtesAsCountCompilation()
    {
        var query = TwoStageQuery();
        var catalog = Catalog(Table(
            "dataverse",
            "contact",
            "contactid",
            "statecode",
            "address1_country",
            "fullname"));

        var countResult = new SegmentCountSqlCompiler().Compile(
            new SegmentCountRequest { Query = query },
            catalog);
        var memberResult = new SegmentCountSqlCompiler().CompileMemberQuery(
            query,
            catalog,
            Options(stageOrder: 1, viewMode: "stage"));

        // Both compilations independently walk the same AST from a fresh
        // CompilationContext. Extracting the exact, byte-identical CTE text
        // for each shared stage from both outputs proves the member query
        // reuses (rather than re-derives with different logic) the same
        // compiled stage bodies used for counts.
        var baseStageCte = ExtractCte(countResult.Sql, "profile_0");
        var filterStageCte = ExtractCte(countResult.Sql, "profile_step_1");

        Assert.Contains(baseStageCte, memberResult.Sql);
        Assert.Contains(filterStageCte, memberResult.Sql);
    }

    private static string ExtractCte(string sql, string cteName)
    {
        var marker = "[" + cteName + "] AS";
        var start = sql.IndexOf(marker, StringComparison.Ordinal);
        Assert.True(start >= 0, $"Expected to find CTE '{cteName}' in SQL.");
        var openParen = sql.IndexOf('(', start);
        var depth = 0;
        var end = openParen;
        for (; end < sql.Length; end++)
        {
            if (sql[end] == '(')
            {
                depth++;
            }
            else if (sql[end] == ')')
            {
                depth--;
                if (depth == 0)
                {
                    end++;
                    break;
                }
            }
        }

        return sql[start..end];
    }

    private static string ExtractLastCteBody(string sql)
    {
        // The CTE *definition* ("[member_N] AS (") appears first in the WITH
        // block; a later reference ("FROM [member_N] AS m") also matches the
        // "[member_" prefix in the outer query, so use the first occurrence.
        var memberIndex = sql.IndexOf("[member_", StringComparison.Ordinal);
        Assert.True(memberIndex >= 0, "Expected a member_ CTE in generated SQL.");
        var openParen = sql.IndexOf('(', memberIndex);
        var selectIndex = sql.IndexOf("SELECT TOP", memberIndex, StringComparison.Ordinal);
        Assert.True(selectIndex > openParen, sql);
        return sql[openParen..selectIndex];
    }

    private static SegmentQueryNode TwoStageQuery()
    {
        return new SegmentQueryNode
        {
            FirstOperand = new SegmentOperand
            {
                Kind = "profile",
                ProfileEntity = "contact",
                BaseLabel = "Active contacts",
                BaseDetail = "PROFILE(contact)",
                Steps =
                [
                    new SegmentFilterStep
                    {
                        Kind = "profile",
                        Label = "Country filter",
                        Detail = "contact.address1_country == 'DE'",
                        Condition = new SegmentCondition
                        {
                            Kind = "predicate",
                            Field = "address1_country",
                            Operator = "==",
                            Value = Json("\"DE\"")
                        }
                    }
                ]
            }
        };
    }

    private static SegmentMemberCompileOptions Options(
        int stageOrder,
        string viewMode,
        string sortField = "fullname",
        bool ascending = true,
        string? search = null,
        string? filterField = null,
        string? filterOperator = null,
        string? filterValue = null,
        int pageSize = 10,
        SegmentMemberCursor? cursor = null)
    {
        return new SegmentMemberCompileOptions(
            stageOrder,
            viewMode,
            sortField,
            ascending,
            search,
            filterField,
            filterOperator,
            filterValue,
            pageSize,
            cursor);
    }

    private static CompiledSegmentMemberQuery Compile(
        SegmentQueryNode query,
        SegmentMemberCompileOptions options,
        params EventTableDescriptor[] tables)
    {
        return new SegmentCountSqlCompiler().CompileMemberQuery(query, Catalog(tables), options);
    }

    private static FabricSegmentCatalogSnapshot Catalog(params EventTableDescriptor[] tables)
    {
        return new FabricSegmentCatalogSnapshot(DateTimeOffset.UtcNow, tables);
    }

    private static EventTableDescriptor Table(string schema, string table, params string[] columns)
    {
        return new EventTableDescriptor(
            schema,
            table,
            columns.ToDictionary(
                IdentifierNormalizer.Normalize,
                column => column,
                StringComparer.OrdinalIgnoreCase));
    }

    private static System.Text.Json.JsonElement Json(string json)
    {
        return System.Text.Json.JsonDocument.Parse(json).RootElement.Clone();
    }
