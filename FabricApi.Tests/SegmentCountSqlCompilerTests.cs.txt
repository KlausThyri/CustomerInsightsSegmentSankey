    {

        Assert.Contains("consent.[msdynmkt_contactpointtype]", result.Sql);
        Assert.Contains("consent.[statecode]", result.Sql);
    }

    private static CompiledSegmentCountQuery CompileEvaluatedConsent(
        int? model, bool withTopic, string value = "WillSend")
    {
        var request = Request(new SegmentOperand
        {
            Kind = "profile",
            ProfileEntity = "contact",
            BaseLabel = "Contacts",
            BaseDetail = "PROFILE(contact)",
            Steps =
            [
                new SegmentFilterStep
                {
                    Kind = "profile",
                    Label = "Consent",
                    Detail = value,
                    Condition = new SegmentCondition
                    {
                        Kind = "consent",
                        ProfileEmailField = "emailaddress1",
                        PurposeId = Guid.Parse("444e8759-da21-f111-8341-6045bde0d602"),
                        TopicId = withTopic
                            ? Guid.Parse("03b9fc9a-da21-f111-8341-6045bde0d602") : null,
                        Channel = "Email",
                        EnforcementModel = model,
                        Value = Json(JsonSerializer.Serialize(value))
                    }
                }
            ]
        });
        var tables = new List<EventTableDescriptor>
        {
            Table("dataverse", "contact", "Id", "statecode", "emailaddress1")
        };
        if (model != 534120002)
        {
            tables.Add(Table("dataverse", "msdynmkt_contactpointconsent4", "Id", "statecode",
                "msdynmkt_contactpointvalue", "msdynmkt_purposeid", "msdynmkt_topicid",
                "msdynmkt_contactpointtype", "msdynmkt_value"));
        }
        return Compile(request, tables.ToArray());
    }

    [Fact]
    public void Compile_SiblingPaths_MemberQueryReusesTheSameRelationshipStages()
    {
        var request = Request(SiblingProfile(
            QualifiedPredicate("tier", "name", "==", Json("\"Gold\""))));
        var catalog = Catalog(
            Table("dataverse", "contact", "contactid", "statecode", "fullname"),
            Table("dataverse", "loyalty", "contactid", "tierid", "programid"),
            Table("dataverse", "tier", "tierid", "name"),
            Table("dataverse", "program", "programid", "name"));
        var compiler = new SegmentCountSqlCompiler();
        var counts = compiler.Compile(request, catalog);
        var members = compiler.CompileMemberQuery(request.Query, catalog,
            new SegmentMemberCompileOptions(1, "stage", "fullname", true,
                null, null, null, null, 10, null));

        var stageStart = counts.Sql.IndexOf("[profile_step_", StringComparison.Ordinal);
        var stageEnd = counts.Sql.IndexOf("\nSELECT", stageStart, StringComparison.Ordinal);
        Assert.True(stageStart >= 0 && stageEnd > stageStart);
        Assert.Contains(counts.Sql[stageStart..stageEnd].TrimEnd(), members.Sql);
    }

    [Fact]
    public void Compile_ProfileStages_UsesCatalogIdentifiersAndValueParameters()
    {
        var request = Request(new SegmentOperand
        {
            Kind = "profile",
            ProfileEntity = "contact",
            BaseLabel = "Active contacts",
            BaseDetail = "PROFILE(contact)",
            Steps =
            [
                new SegmentFilterStep
                {
                    Kind = "profile",
                    Label = "Country filter",
                    Detail = "contact.address1_country == 'DE'",
                    Condition = Predicate("contact.address1_country", "==", Json("\"DE\"")),
                    Presentation = new SegmentPresentationHint
                    {
                        Entity = "contact",
                        Field = "address1_country",
                        Values = ["DE"],
                        FieldAsTitle = true
                    }
                }
            ]
        });

        var result = Compile(
            request,
            Table("dataverse", "contact", "contactid", "statecode", "address1_country"));

        Assert.Contains("FROM [dataverse].[contact] AS p", result.Sql);
        Assert.Contains("TRY_CONVERT(uniqueidentifier, p.[contactid])", result.Sql);
        Assert.Contains("p.[statecode] = @p0", result.Sql);
        Assert.Contains("p.[address1_country] = @p1", result.Sql);
        Assert.DoesNotContain("'DE'", result.Sql);
        Assert.Contains("COUNT_BIG(*) AS StageCount", result.Sql);
        Assert.EndsWith("ORDER BY StageOrder;", result.Sql);
        Assert.Equal(
            [0, "DE"],
            result.Parameters.Select(parameter => parameter.Value).ToArray());
        Assert.Equal(2, result.Stages.Count);
        Assert.Equal("Country filter", result.Stages[1].Label);
        Assert.Equal("contact", result.Stages[1].Presentation?.Entity);
        Assert.Equal("address1_country", result.Stages[1].Presentation?.Field);
        Assert.Equal(["DE"], result.Stages[1].Presentation?.Values);
        Assert.True(result.Stages[1].Presentation?.FieldAsTitle == true);
        Assert.Collection(
            result.SourceTables,
            source =>
            {
                Assert.Equal("dataverse", source.Schema);
                Assert.Equal("contact", source.Table);
            });
    }

        {

            BaseLabel = "Contacts",

            Steps =
            [
                new SegmentFilterStep
                {
                    Kind = "relationship",
                    Label = "Marketing list",

                    Condition = QualifiedPredicate(
                        "list_1",

                    [
                        new SegmentRelationshipHop
                        {
                            Alias = "list_1",
                            RelatedEntity = "list",
                            SourcePrimaryAttribute = "contactid",
                            RelatedPrimaryAttribute = "listid",
                            IntersectEntity = "listmember",
                            SourceIntersectAttribute = "entityid",
                            RelatedIntersectAttribute = "listid",
                            IsOptional = true
                        }
                    ]
                }
            ]
        };

        var result = Compile(
            Request(profile),

            Table("dataverse", "listmember", "entityid", "listid"),

        Assert.Contains("r.[listid] = @p1", result.Sql);

    [Fact]
    public void Compile_AllSetOperations_UsesStagedSetCtes()
    {
        var id = Guid.Parse("8d78b7a9-9f53-46a2-b39a-ec9eec276c20");
        var request = Request(Static("Initial", id));
        request.Query.SetOperations =
        [
            Set("UNION", "Union", Static("Union operand", Guid.NewGuid())),
            Set("INTERSECT", "Intersection", Static("Intersect operand", id)),
            Set(
                "EXCEPT",
                "Exclusion",
                new SegmentOperand
                {
                    Kind = "query",
                    ProfileEntity = "contact",
                    BaseLabel = "Nested",
                    BaseDetail = "Nested query",
                    Query = Request(Static("Nested static", Guid.NewGuid())).Query
                })
        ];

        var result = Compile(request);

        Assert.Contains(Environment.NewLine + "    UNION" + Environment.NewLine, result.Sql);
        Assert.Contains(Environment.NewLine + "    INTERSECT" + Environment.NewLine, result.Sql);
        Assert.Contains(Environment.NewLine + "    EXCEPT" + Environment.NewLine, result.Sql);
        Assert.Equal(
            ["Initial", "Union", "Intersection", "Exclusion"],
            result.Stages.Select(stage => stage.Label).ToArray());

    }

    private static SegmentFilterStep Relationship(
        string type,
        string label,
        string detail,
        SegmentCondition condition)
    {
        return new SegmentFilterStep
        {
            Kind = "relationship",
            IsOptional = type == "RELATEOPTIONAL",
            RelatedEntity = "account",
            ProfileAttribute = "parentcustomerid",
            RelatedAttribute = "accountid",
            Label = label,
            Detail = detail,
            Condition = condition
        };
    }

    private static SegmentSetOperation Set(
        string operation,
        string label,
        SegmentOperand operand)
    {
        return new SegmentSetOperation
        {
            Operator = operation,
            Operand = operand,
            Label = label,
            Detail = label + " detail"
        };
    }

    private static SegmentOperand Static(string label, params Guid[] ids)
    {
        return new SegmentOperand
        {
            Kind = "static",
            ProfileEntity = "contact",
            ProfileIds = ids.ToList(),
            BaseLabel = label,
            BaseDetail = label + " detail"
        };
    }

    private static SegmentCountRequest Request(SegmentOperand operand)
    {
        return new SegmentCountRequest
        {
            Query = new SegmentQueryNode
            {
                FirstOperand = operand
            }
        };
    }

    private static SegmentCondition Predicate(
        string field,
        string operation,
        JsonElement value)
    {
        return new SegmentCondition
        {
            Kind = "predicate",
            Field = field,
            Operator = operation,
            Value = value
        };
    }

    private static SegmentCondition QualifiedPredicate(
        string qualifier,
        string field,
        string operation,
        JsonElement value)
    {
        var condition = Predicate(field, operation, value);
        condition.Qualifier = qualifier;
        return condition;
    }

    private static JsonElement Json(string json)
    {
        return JsonDocument.Parse(json).RootElement.Clone();
    }

    private static CompiledSegmentCountQuery Compile(
        SegmentCountRequest request,
        params EventTableDescriptor[] tables)
    {
        return new SegmentCountSqlCompiler().Compile(request, Catalog(tables));
    }

    private static FabricSegmentCatalogSnapshot Catalog(
        params EventTableDescriptor[] tables)
    {
        return new FabricSegmentCatalogSnapshot(DateTimeOffset.UtcNow, tables);
    }

    private static EventTableDescriptor Table(
        string schema,
        string table,
        params string[] columns)
    {
        return new EventTableDescriptor(
            schema,
            table,
            columns.ToDictionary(
                IdentifierNormalizer.Normalize,
                column => column,
                StringComparer.OrdinalIgnoreCase));
    }
