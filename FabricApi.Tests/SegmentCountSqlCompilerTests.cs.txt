    [Fact]
    public void Ast_CamelCaseUnionShape_DeserializesOperandAndSteps()
    {
        const string json = """
            {
              "query": {
                "firstOperand": {
                  "kind": "profile",
                  "profileEntity": "contact",
                  "baseLabel": "Contacts",
                  "baseDetail": "PROFILE(contact)",
                  "steps": [{
                    "kind": "profile",
                    "label": "Active",
                    "detail": "statecode == 0",
                    "condition": {
                      "kind": "predicate",
                      "field": "statecode",
                      "operator": "==",
                      "value": 0
                    }
                  }]
                },
                "setOperations": []
              }
            }
            """;

        var request = JsonSerializer.Deserialize<SegmentCountRequest>(
            json,
            new JsonSerializerOptions(JsonSerializerDefaults.Web));

        var profile = Assert.IsType<SegmentOperand>(request!.Query.FirstOperand);
        Assert.Equal("profile", profile.Kind);
        Assert.Equal("profile", Assert.Single(profile.Steps).Kind);
    }

    [Fact]
    public void Compile_ProfileStages_UsesCatalogIdentifiersAndValueParameters()
    {
        var request = Request(new SegmentOperand
        {
            Kind = "profile",
            ProfileEntity = "contact",
            BaseLabel = "Active contacts",
            BaseDetail = "PROFILE(contact)",
            Steps =
            [
                new SegmentFilterStep
                {
                    Kind = "profile",
                    Label = "Country filter",
                    Detail = "contact.address1_country == 'DE'",
                    Condition = Predicate("contact.address1_country", "==", Json("\"DE\""))
                }
            ]
        });

        var result = Compile(
            request,
            Table("dataverse", "contact", "contactid", "statecode", "address1_country"));

        Assert.Contains("FROM [dataverse].[contact] AS p", result.Sql);
        Assert.Contains("TRY_CONVERT(uniqueidentifier, p.[contactid])", result.Sql);
        Assert.Contains("p.[statecode] = @p0", result.Sql);
        Assert.Contains("p.[address1_country] = @p1", result.Sql);
        Assert.DoesNotContain("'DE'", result.Sql);
        Assert.Contains("COUNT_BIG(*) AS StageCount", result.Sql);
        Assert.EndsWith("ORDER BY StageOrder;", result.Sql);
        Assert.Equal(
            [0, "DE"],
            result.Parameters.Select(parameter => parameter.Value).ToArray());
        Assert.Equal(2, result.Stages.Count);
        Assert.Equal("Country filter", result.Stages[1].Label);
        Assert.Collection(
            result.SourceTables,
            source =>
            {
                Assert.Equal("dataverse", source.Schema);
                Assert.Equal("contact", source.Table);
            });
    }

        var request = Request(new SegmentOperand
        {

    [Fact]
    public void Compile_RelationshipRootNot_DistinguishesRelateAndOptional()
    {
        var notPreferred = new SegmentCondition
        {
            Kind = "not",
            Children = [Predicate("account.category", "==", Json("\"preferred\""))]
        };
        var profile = new SegmentOperand
        {
            Kind = "profile",
            ProfileEntity = "contact",
            BaseLabel = "Contacts",
            BaseDetail = "PROFILE(contact)",
            Steps =
            [
                Relationship(
                    "RELATE",
                    "Required relationship",
                    "Required root NOT",
                    notPreferred),
                Relationship(
                    "RELATEOPTIONAL",
                    "Optional relationship",
                    "Optional root NOT",
                    notPreferred)
            ]
        };

        var result = Compile(
            Request(profile),
            Table("dataverse", "contact", "contactid", "statecode", "parentcustomerid"),
            Table("dataverse", "account", "accountid", "category"));

        Assert.Contains("EXISTS (SELECT 1", result.Sql);
        Assert.Contains("AND NOT (r.[category] = @p1)", result.Sql);
        Assert.Contains("NOT EXISTS (SELECT 1", result.Sql);
        Assert.Contains("AND r.[category] = @p2", result.Sql);
        Assert.Contains("r.[accountid] = p.[parentcustomerid]", result.Sql);
        Assert.Equal(3, result.Stages.Count);
        Assert.Equal(2, result.SourceTables.Count);
    }

    [Fact]
    public void Compile_NestedRelationshipPath_JoinsEveryHopAndFiltersInnermostTable()
    {
        var profile = new SegmentOperand
        {
            Kind = "profile",
            ProfileEntity = "contact",
            BaseLabel = "Contacts",
            BaseDetail = "PROFILE(contact)",
            Steps =
            [
                new SegmentFilterStep
                {
                    Kind = "relationship",
                    Label = "Loyalty points",
                    Detail = "Nested Customer Insights measure",
                    IsOptional = true,
                    Condition = new SegmentCondition
                    {
                        Kind = "and",
                        Children =
                        [
                            QualifiedPredicate(
                                "msdynci_loyalitypoints_1",
                                "statecode",
                                "==",
                                Json("0")),
                            QualifiedPredicate(
                                "msdynci_loyalitypoints_1__1",
                                "msdynci_sumofpoints",
                                ">=",
                                Json("100"))
                        ]
                    },
                    Relationships =
                    [
                        new SegmentRelationshipHop
                        {
                            Alias = "CustomerProfile.msdynci_loyalitypoints_1",
                            RelatedEntity = "msdynci_customerprofile",
                            SourceAttribute = "msdynci_customerprofileid",
                            RelatedAttribute = "msdynci_customerprofileid",
                            IsOptional = true
                        },
                        new SegmentRelationshipHop
                        {
                            Alias = "msdynci_loyalitypoints_1__1",
                            RelatedEntity = "msdynci_loyalitypoints",
                            SourceAttribute = "msdynci_customerprofileid",
                            RelatedAttribute = "msdynci_customerprofileid",
                            IsOptional = true
                        }
                    ]
                }
            ]
        };

        var result = Compile(
            Request(profile),
            Table(
                "dataverse",
                "contact",
                "contactid",
                "statecode",
                "msdynci_customerprofileid"),
            Table(
                "dataverse",
                "msdynci_customerprofile",
                "msdynci_customerprofileid",
                "statecode"),
            Table(
                "dataverse",
                "msdynci_loyalitypoints",
                "msdynci_customerprofileid",
                "msdynci_sumofpoints"));

        Assert.Contains(
            "FROM [dataverse].[msdynci_customerprofile] AS r1",
            result.Sql);
        Assert.Contains(
            "INNER JOIN [dataverse].[msdynci_loyalitypoints] AS r2 ON " +
            "r2.[msdynci_customerprofileid] = r1.[msdynci_customerprofileid]",
            result.Sql);
        Assert.Contains(
            "WHERE r1.[msdynci_customerprofileid] = p.[msdynci_customerprofileid]",
            result.Sql);
        Assert.Contains("r1.[statecode] = @p1", result.Sql);
        Assert.Contains("r2.[msdynci_sumofpoints] >= @p2", result.Sql);
        Assert.Equal(3, result.SourceTables.Count);
    }

    [Fact]
    public void Compile_AllSetOperations_UsesStagedSetCtes()
    {
        var id = Guid.Parse("8d78b7a9-9f53-46a2-b39a-ec9eec276c20");
        var request = Request(Static("Initial", id));
        request.Query.SetOperations =
        [
            Set("UNION", "Union", Static("Union operand", Guid.NewGuid())),
            Set("INTERSECT", "Intersection", Static("Intersect operand", id)),
            Set(
                "EXCEPT",
                "Exclusion",
                new SegmentOperand
                {
                    Kind = "query",
                    ProfileEntity = "contact",
                    BaseLabel = "Nested",
                    BaseDetail = "Nested query",
                    Query = Request(Static("Nested static", Guid.NewGuid())).Query
                })
        ];

        var result = Compile(request);

        Assert.Contains(Environment.NewLine + "    UNION" + Environment.NewLine, result.Sql);
        Assert.Contains(Environment.NewLine + "    INTERSECT" + Environment.NewLine, result.Sql);
        Assert.Contains(Environment.NewLine + "    EXCEPT" + Environment.NewLine, result.Sql);
        Assert.Equal(
            ["Initial", "Union", "Intersection", "Exclusion"],
            result.Stages.Select(stage => stage.Label).ToArray());
        Assert.Contains("UNION ALL", result.Sql);
    }

    [Fact]
    public void Compile_Interaction_UsesAliasesHavingWindowAndParameters()
    {
        var request = Request(new SegmentOperand
        {
            Kind = "interaction",
            ProfileEntity = "contact",
            EventLogicalName = "msdynmkt_emaillinkclicked",
            EntityIdField = "msdynmkt_entityid",
            BaseLabel = "Clicked",
            BaseDetail = "Interaction click",
            Filter = Predicate("msdynmkt_messagetemplateid", "==", Json("\"newsletter\"")),
            Having = new SegmentHaving
            {
                Metric = "Count",
                Operator = ">=",
                Threshold = 2,
                WindowFunction = "UTCDAYS",
                WindowValue = 28
            }
        });

        var result = Compile(
            request,
            Table(
                "journeys",
                "RedirectLinkClicked",
                "ProfileId",
                "ContactId",
                "ProfileType",
                "MessageId",
                "Timestamp"));

        Assert.Contains("FROM [journeys].[RedirectLinkClicked] AS e", result.Sql);
        Assert.Contains("COALESCE([ProfileId], [ContactId])", result.Sql);
        Assert.Contains("[MessageId] = @p1", result.Sql);
        Assert.Contains("DATEADD(day, -@p2, SYSUTCDATETIME())", result.Sql);
        Assert.Contains("HAVING COUNT_BIG(*) >= @p3", result.Sql);
        Assert.DoesNotContain("newsletter", result.Sql);
        Assert.Equal(
            ["contact", "newsletter", 28, 2L],
            result.Parameters.Select(parameter => parameter.Value).ToArray());
    }

    private static SegmentFilterStep Relationship(
        string type,
        string label,
        string detail,
        SegmentCondition condition)
    {
        return new SegmentFilterStep
        {
            Kind = "relationship",
            IsOptional = type == "RELATEOPTIONAL",
            RelatedEntity = "account",
            ProfileAttribute = "parentcustomerid",
            RelatedAttribute = "accountid",
            Label = label,
            Detail = detail,
            Condition = condition
        };
    }

    private static SegmentSetOperation Set(
        string operation,
        string label,
        SegmentOperand operand)
    {
        return new SegmentSetOperation
        {
            Operator = operation,
            Operand = operand,
            Label = label,
            Detail = label + " detail"
        };
    }

    private static SegmentOperand Static(string label, params Guid[] ids)
    {
        return new SegmentOperand
        {
            Kind = "static",
            ProfileEntity = "contact",
            ProfileIds = ids.ToList(),
            BaseLabel = label,
            BaseDetail = label + " detail"
        };
    }

    private static SegmentCountRequest Request(SegmentOperand operand)
    {
        return new SegmentCountRequest
        {
            Query = new SegmentQueryNode
            {
                FirstOperand = operand
            }
        };
    }

    private static SegmentCondition Predicate(
        string field,
        string operation,
        JsonElement value)
    {
        return new SegmentCondition
        {
            Kind = "predicate",
            Field = field,
            Operator = operation,
            Value = value
        };
    }

    private static SegmentCondition QualifiedPredicate(
        string qualifier,
        string field,
        string operation,
        JsonElement value)
    {
        var condition = Predicate(field, operation, value);
        condition.Qualifier = qualifier;
        return condition;
    }

    private static JsonElement Json(string json)
    {
        return JsonDocument.Parse(json).RootElement.Clone();
    }

    private static CompiledSegmentCountQuery Compile(
        SegmentCountRequest request,
        params EventTableDescriptor[] tables)
    {
        return new SegmentCountSqlCompiler().Compile(request, Catalog(tables));
    }

    private static FabricSegmentCatalogSnapshot Catalog(
        params EventTableDescriptor[] tables)
    {
        return new FabricSegmentCatalogSnapshot(DateTimeOffset.UtcNow, tables);
    }

    private static EventTableDescriptor Table(
        string schema,
        string table,
        params string[] columns)
    {
        return new EventTableDescriptor(
            schema,
            table,
            columns.ToDictionary(
                IdentifierNormalizer.Normalize,
                column => column,
                StringComparer.OrdinalIgnoreCase));
    }
