#if FABRIC_API
using CustomerInsightsSegmentSankey.FabricApi;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Logging.Abstractions;
using Xunit;

namespace CustomerInsightsSegmentSankey.FabricApi.Tests;

/// <summary>
/// Covers <see cref="SegmentDependencyValidationQueue"/>: TTL-based
/// dedup/debounce, <see cref="ISegmentDependencyValidationQueue.MarkValidated"/>
/// suppressing redundant re-enqueue, background failures being logged rather
/// than thrown, and a clean, drain-then-stop shutdown.
/// </summary>
public sealed class SegmentDependencyValidationQueueTests
{
    [Fact]
    public async Task Enqueue_reconciles_the_requested_tables_exactly_once()
    {
        var reconciler = new RecordingReconciler();
        using var queue = CreateQueue(reconciler, out var scopeFactory);
        await queue.StartAsync(CancellationToken.None);

        queue.Enqueue(["account"]);
        await reconciler.WaitForCallAsync();

        Assert.Equal(["account"], reconciler.Calls.Single());
        await queue.StopAsync(CancellationToken.None);
    }

    [Fact]
    public async Task Enqueue_within_the_success_TTL_does_not_reconcile_again()
    {
        var reconciler = new RecordingReconciler();
        using var queue = CreateQueue(reconciler, out _);
        await queue.StartAsync(CancellationToken.None);

        queue.Enqueue(["account"]);
        await reconciler.WaitForCallAsync();
        queue.Enqueue(["account"]);

        // Give the background loop a moment to (incorrectly) process a
        // second batch if debouncing were broken.
        await Task.Delay(TimeSpan.FromMilliseconds(200));

        Assert.Single(reconciler.Calls);
        await queue.StopAsync(CancellationToken.None);
    }

    [Fact]
    public async Task MarkValidated_suppresses_a_subsequent_enqueue_without_reconciling()
    {
        var reconciler = new RecordingReconciler();
        using var queue = CreateQueue(reconciler, out _);
        await queue.StartAsync(CancellationToken.None);

        queue.MarkValidated(["account"]);
        queue.Enqueue(["account"]);
        await Task.Delay(TimeSpan.FromMilliseconds(200));

        Assert.Empty(reconciler.Calls);
        await queue.StopAsync(CancellationToken.None);
    }

    [Fact]
    public async Task Failed_reconciliation_is_logged_and_does_not_crash_the_queue()
    {
        var reconciler = new RecordingReconciler(shouldFail: true);
        var logger = new RecordingLogger<SegmentDependencyValidationQueue>();
        using var queue = CreateQueue(reconciler, out _, logger);
        await queue.StartAsync(CancellationToken.None);

        queue.Enqueue(["account"]);
        await reconciler.WaitForCallAsync();
        await Task.Delay(TimeSpan.FromMilliseconds(100));

        Assert.Single(logger.Warnings);
        Assert.Contains("account", logger.Warnings[0]);

        // A failed table is not marked validated, so a later request for the
        // same table can still retry reconciliation instead of being stuck.
        queue.Enqueue(["account"]);
        await reconciler.WaitForCallAsync(expectedCallCount: 2);
        Assert.Equal(2, reconciler.Calls.Count);

        await queue.StopAsync(CancellationToken.None);
    }

    [Fact]
    public async Task StopAsync_drains_an_already_enqueued_batch_before_shutting_down()
    {
        var reconciler = new RecordingReconciler();
        using var queue = CreateQueue(reconciler, out _);
        await queue.StartAsync(CancellationToken.None);

        queue.Enqueue(["account"]);
        await reconciler.WaitForCallAsync();

        await queue.StopAsync(CancellationToken.None);

        Assert.Single(reconciler.Calls);
    }

    [Fact]
    public async Task Concurrent_enqueues_for_the_same_table_reconcile_it_exactly_once()
    {
        // Every concurrent caller races the same "not already pending, not
        // within the success TTL" check; only one may atomically win the
        // claim on the shared table. A gate holds the very first
        // reconciliation in flight while the remaining concurrent Enqueue
        // calls race it, so a "check pending, then set pending" race (as
        // opposed to an atomic TryAdd) would show up as more than one call.
        var reconciler = new RecordingReconciler { Gate = new TaskCompletionSource() };
        using var queue = CreateQueue(reconciler, out _);
        await queue.StartAsync(CancellationToken.None);

        var barrier = new Barrier(8);
        var enqueuers = Enumerable.Range(0, 8).Select(_ => Task.Run(() =>
        {
            barrier.SignalAndWait();
            queue.Enqueue(["account"]);
        }));
        await Task.WhenAll(enqueuers);

        reconciler.Gate.SetResult();
        await reconciler.WaitForCallAsync();
        await Task.Delay(TimeSpan.FromMilliseconds(200));

        Assert.Single(reconciler.Calls);

        await queue.StopAsync(CancellationToken.None);
    }

    [Fact]
    public async Task StopAsync_drains_in_flight_and_still_queued_work_without_hanging()
    {
        var reconciler = new RecordingReconciler { Gate = new TaskCompletionSource() };
        using var queue = CreateQueue(reconciler, out _);
        await queue.StartAsync(CancellationToken.None);

        // "account" starts reconciling and blocks on the gate (in flight);
        // "contact" is claimed and written to the channel behind it (still
        // queued, not yet started) by the time StopAsync is invoked.
        queue.Enqueue(["account"]);
        await reconciler.WaitForCallAsync();
        queue.Enqueue(["contact"]);

        var stopTask = queue.StopAsync(CancellationToken.None);
        await Task.Delay(TimeSpan.FromMilliseconds(150));
        Assert.False(stopTask.IsCompleted);

        reconciler.Gate.SetResult();

        var completed = await Task.WhenAny(stopTask, Task.Delay(TimeSpan.FromSeconds(5)));
        Assert.True(ReferenceEquals(completed, stopTask), "StopAsync hung instead of draining queued/in-flight work.");
        await stopTask;

        Assert.Equal(2, reconciler.Calls.Count);
        Assert.Contains(reconciler.Calls, call => call.SequenceEqual(new[] { "account" }));
        Assert.Contains(reconciler.Calls, call => call.SequenceEqual(new[] { "contact" }));
    }

    private static SegmentDependencyValidationQueue CreateQueue(
        RecordingReconciler reconciler,
        out IServiceScopeFactory scopeFactory,
        ILogger<SegmentDependencyValidationQueue>? logger = null)
    {
        var services = new ServiceCollection();
        services.AddSingleton<IFabricDependencyReconciler>(_ => reconciler);
        var provider = services.BuildServiceProvider();
        scopeFactory = provider.GetRequiredService<IServiceScopeFactory>();
        return new SegmentDependencyValidationQueue(
            scopeFactory,
            logger ?? NullLogger<SegmentDependencyValidationQueue>.Instance);
    }

    private sealed class RecordingReconciler(bool shouldFail = false) : IFabricDependencyReconciler
    {
        public List<List<string>> Calls { get; } = [];

        /// <summary>
        /// Optional gate every call awaits (with <see cref="CancellationToken.None"/>,
        /// matching production) before returning, letting tests observe and
        /// control an "in-flight" reconciliation.
        /// </summary>
        public TaskCompletionSource? Gate { get; set; }

        public async Task<FabricDependencyResponse> ReconcileAsync(
            IReadOnlyCollection<string> tables,
            CancellationToken cancellationToken)
        {
            lock (Calls)
            {
                Calls.Add(tables.ToList());
            }

            if (Gate is not null)
            {
                await Gate.Task;
            }

            if (shouldFail)
            {
                throw new InvalidOperationException("Simulated reconciliation failure.");
            }

            return new FabricDependencyResponse
            {
                RequiredTables = tables.ToList(),
                CatalogReady = true
            };
        }

        public async Task WaitForCallAsync(int expectedCallCount = 1)
        {
            var deadline = DateTime.UtcNow.Add(TimeSpan.FromSeconds(5));
            while (DateTime.UtcNow < deadline)
            {
                lock (Calls)
                {
                    if (Calls.Count >= expectedCallCount)
                    {
                        return;
                    }
                }

                await Task.Delay(TimeSpan.FromMilliseconds(25));
            }

            throw new TimeoutException("Expected the background reconciler to be invoked.");
        }
    }

    private sealed class RecordingLogger<T> : ILogger<T>
    {
        public List<string> Warnings { get; } = [];

        public IDisposable? BeginScope<TState>(TState state) where TState : notnull => null;

        public bool IsEnabled(LogLevel logLevel) => true;

        public void Log<TState>(
            LogLevel logLevel,
            EventId eventId,
            TState state,
            Exception? exception,
            Func<TState, Exception?, string> formatter)
        {
            if (logLevel == LogLevel.Warning)
            {
                lock (Warnings)
                {
                    Warnings.Add(formatter(state, exception));
                }
            }
        }
    }
}
#endif
