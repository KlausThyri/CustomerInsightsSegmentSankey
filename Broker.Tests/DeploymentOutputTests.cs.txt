using CustomerInsightsSegmentSankey.Broker;
using System.Text.Json;
using Xunit;

namespace CustomerInsightsSegmentSankey.Broker.Tests;

public class DeploymentOutputTests
{
    [Fact]
    public void PublishesTheManagedIdentityBeforeTheFabricRoleStepReadsIt()
    {
        var outputs = new Dictionary<string, string?>
        {
            ["managedIdentityPrincipalId"] = "11111111-2222-3333-4444-555555555555",
            ["webAppUrl"] = "https://segment-preview.azurewebsites.net/api/"
        };

        AzureFabricProvisioningExecutor.ApplyDeploymentOutputs(outputs, "segment-preview");

        Assert.Equal("11111111-2222-3333-4444-555555555555", outputs["principalId"]);
    }

    [Fact]
    public void KeepsAnAlreadyReportedPrincipalWhenTheTemplateNamesItDirectly()
    {
        var outputs = new Dictionary<string, string?>
        {
            ["principalId"] = "aaaaaaaa-bbbb-cccc-dddd-eeeeeeeeeeee",
            ["webAppUrl"] = "https://segment-preview.azurewebsites.net/api/"
        };

        AzureFabricProvisioningExecutor.ApplyDeploymentOutputs(outputs, "segment-preview");

        Assert.Equal("aaaaaaaa-bbbb-cccc-dddd-eeeeeeeeeeee", outputs["principalId"]);
    }

    [Fact]
    public void UsesTheReportedWebAppUrlAsTheApiBaseUrl()
    {
        var outputs = new Dictionary<string, string?>
        {
            ["webAppUrl"] = "https://segment-preview.azurewebsites.net/api/"
        };

        AzureFabricProvisioningExecutor.ApplyDeploymentOutputs(outputs, "segment-preview");

        Assert.Equal("https://segment-preview.azurewebsites.net/api/", outputs["apiBaseUrl"]);
    }

    [Fact]
    public void FallsBackToTheWebAppNameAndStillEndsInTheApiSegment()
    {
        var outputs = new Dictionary<string, string?>();

        AzureFabricProvisioningExecutor.ApplyDeploymentOutputs(outputs, "segment-preview");

        Assert.Equal("https://segment-preview.azurewebsites.net/api/", outputs["apiBaseUrl"]);
    }

    [Theory]
    [InlineData("https://segment-preview.azurewebsites.net", "https://segment-preview.azurewebsites.net/api/")]
    [InlineData("https://segment-preview.azurewebsites.net/", "https://segment-preview.azurewebsites.net/api/")]
    [InlineData("https://segment-preview.azurewebsites.net/api", "https://segment-preview.azurewebsites.net/api/")]
    [InlineData("segment-preview.azurewebsites.net", "https://segment-preview.azurewebsites.net/api/")]
    public void AlwaysReturnsTheApiRoot(string reported, string expected)
    {
        Assert.Equal(expected, AzureFabricProvisioningExecutor.NormalizeApiBaseUrl(reported));
    }

    [Fact]
    public void RefusesAnEmptyWebAppUrl()
    {
        Assert.Throws<InvalidOperationException>(() => AzureFabricProvisioningExecutor.NormalizeApiBaseUrl("  "));
    }

    [Theory]
    [InlineData("Tables", true)]
    [InlineData("/Tables/", true)]
    [InlineData("Tables/dataverse", false)]
    public void DataverseDiscoveryAcceptsOnlyRootSourceShortcuts(
        string path,
        bool expected)
    {
        using var json = JsonDocument.Parse(
            $$"""
            {
              "path": "{{path}}",
              "name": "contact",
              "target": {
                "dataverse": {
                  "connectionId": "11111111-1111-1111-1111-111111111111",
                  "deltaLakeFolder": "managed/contact",
                  "environmentDomain": "https://contoso.crm4.dynamics.com"
                }
              }
            }
            """);

        var found = AzureFabricProvisioningExecutor.IsRootDataverseShortcut(
            json.RootElement,
            "contoso.crm4.dynamics.com",
            out var name,
            out var connectionId,
            out var deltaLakeFolder);

        Assert.Equal(expected, found);
        if (expected)
        {
            Assert.Equal("contact", name);
            Assert.Equal(
                "11111111-1111-1111-1111-111111111111",
                connectionId);
            Assert.Equal("managed/contact", deltaLakeFolder);
        }
    }
}
