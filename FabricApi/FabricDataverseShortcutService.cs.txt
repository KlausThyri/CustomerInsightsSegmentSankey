using System.Net.Http.Headers;
using System.Net.Http.Json;
using System.Net;
using System.Text.Json;
using System.Text.RegularExpressions;
using Azure.Core;
using Azure.Identity;

namespace CustomerInsightsSegmentSankey.FabricApi;

internal sealed class FabricDataverseShortcutService
{
    private const int MaximumTablesPerRequest = 20;
    private const int MaximumCreateAttempts = 5;
    private static readonly Regex SafeTableName = new(
        "^[A-Za-z][A-Za-z0-9_]*$",
        RegexOptions.CultureInvariant);

    private readonly HttpClient httpClient;
    private readonly TokenCredential credential;
    private readonly string workspaceId;
    private readonly string lakehouseId;
    private readonly string dataverseLakehouseId;
    private readonly string sqlEndpointId;

    public FabricDataverseShortcutService(HttpClient httpClient)
    {
        this.httpClient = httpClient;
        credential = new DefaultAzureCredential();
        workspaceId = ReadRequiredSetting("FABRIC_WORKSPACE_ID");
        lakehouseId = ReadRequiredSetting("FABRIC_SERVING_LAKEHOUSE_ID");
        dataverseLakehouseId = ReadRequiredSetting("FABRIC_DATAVERSE_LAKEHOUSE_ID");
        sqlEndpointId = ReadRequiredSetting("FABRIC_SQL_DATABASE");
    }

    public async Task<FabricDependencyResponse> EnsureAsync(
        FabricDependencyRequest request,
        CancellationToken cancellationToken)
    {
        var required = (request.DataverseTables ?? [])
            .Where(name => !string.IsNullOrWhiteSpace(name))
            .Select(name => name.Trim())
            .Distinct(StringComparer.OrdinalIgnoreCase)
            .OrderBy(name => name, StringComparer.OrdinalIgnoreCase)
            .ToList();
        if (required.Count > MaximumTablesPerRequest)
        {
            throw new FabricQueryException(
                "dependency_limit_exceeded",
                $"At most {MaximumTablesPerRequest} Dataverse tables can be added per request.");
        }

        var unsafeName = required.FirstOrDefault(name => !SafeTableName.IsMatch(name));
        if (unsafeName is not null)
        {
            throw new FabricQueryException(
                "invalid_table_name",
                $"The Dataverse table name '{unsafeName}' is invalid.");
        }

        var token = await credential.GetTokenAsync(
            new TokenRequestContext(["https://api.fabric.microsoft.com/.default"]),
            cancellationToken);
        var existing = await ListExistingTargetsAsync(token.Token, cancellationToken);
        var sourceTargets = await ListSourceTargetsAsync(token.Token, cancellationToken);
        var missingSources = required
            .Where(table => !sourceTargets.ContainsKey(table))
            .ToList();
        if (missingSources.Count > 0)
        {
            throw new FabricQueryException(
                "shortcut_source_missing",
                "These tables are not available as Dataverse Managed Lake shortcuts: " +
                string.Join(", ", missingSources) + ". In Power Apps, open Link to Microsoft " +
                "Fabric > Manage tables, add the tables, and wait for their initial sync to " +
                "finish before retrying the segment preview.");
        }

        var tablesToAdd = required
            .Where(table =>
                !existing.TryGetValue(table, out var existingTarget) ||
                existingTarget is null ||
                !TargetsMatch(existingTarget, sourceTargets[table]))
            .ToList();

        var added = new List<string>();
        foreach (var table in tablesToAdd)
        {
            var sourceTarget = sourceTargets[table];
            for (var attempt = 1; attempt <= MaximumCreateAttempts; attempt++)
            {
                using var createRequest = new HttpRequestMessage(
                    HttpMethod.Post,
                    BuildCreateShortcutUri());
                createRequest.Headers.Authorization =
                    new AuthenticationHeaderValue("Bearer", token.Token);
                createRequest.Content = CreateShortcutContent(table, sourceTarget);
                using var createResponse = await httpClient.SendAsync(
                    createRequest,
                    cancellationToken);
                var createBody = await createResponse.Content.ReadAsStringAsync(
                    cancellationToken);
                if (createResponse.IsSuccessStatusCode)
                {
                    break;
                }

                if (IsTransientCreateStatus(createResponse.StatusCode) &&
                    attempt < MaximumCreateAttempts)
                {
                    await Task.Delay(
                        TimeSpan.FromSeconds(Math.Min(16, 1 << (attempt - 1))),
                        cancellationToken);
                    continue;
                }

                throw new FabricQueryException(
                    "shortcut_create_failed",
                    CreateFailureMessage(table, createResponse.StatusCode),
                    createBody);
            }

            added.Add(table);
            existing[table] = sourceTarget;
        }

        return new FabricDependencyResponse
        {
            RequiredTables = required,
            AddedTables = added,
            ExistingTables = required
                .Where(table => !added.Contains(table, StringComparer.OrdinalIgnoreCase))
                .ToList()
        };
    }

    internal static bool IsTransientCreateStatus(HttpStatusCode statusCode) =>
        statusCode is HttpStatusCode.Unauthorized or
            HttpStatusCode.Forbidden or
            HttpStatusCode.RequestTimeout or
            HttpStatusCode.TooManyRequests ||
        (int)statusCode >= 500;

    internal static string CreateFailureMessage(
        string table,
        HttpStatusCode statusCode)
    {
        var prefix = $"The Dataverse table '{table}' could not be added to Fabric. ";
        if (statusCode is HttpStatusCode.BadRequest or HttpStatusCode.NotFound)
        {
            return prefix +
                $"Ensure '{table}' is selected under Tables > Analyze > Link to Microsoft Fabric " +
                "and wait until it appears in the linked Lakehouse before running Setup again.";
        }

        if (statusCode is HttpStatusCode.Unauthorized or HttpStatusCode.Forbidden)
        {
            return prefix +
                "Fabric has not accepted the Web App managed identity's workspace or Dataverse " +
                "connection access. Wait a few minutes and retry Setup. If it still fails, verify " +
                "the workspace role and that the cloud connection can be shared.";
        }

        return prefix + "Retry Setup after Fabric is available.";
    }

    public async Task<IReadOnlyCollection<string>> GetExistingAsync(
        CancellationToken cancellationToken)
    {
        var token = await credential.GetTokenAsync(
            new TokenRequestContext(["https://api.fabric.microsoft.com/.default"]),
            cancellationToken);
        return (await ListExistingTargetsAsync(token.Token, cancellationToken)).Keys;
    }

    public async Task RefreshSqlMetadataAsync(
        IReadOnlyCollection<string> tables,
        IReadOnlyCollection<string> newlyCreatedTables,
        CancellationToken cancellationToken)
    {
        if (tables.Count == 0)
        {
            return;
        }

        using var refreshTimeout =
            CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        refreshTimeout.CancelAfter(TimeSpan.FromSeconds(12));
        var refreshToken = refreshTimeout.Token;
        AccessToken token;
        try
        {
            token = await credential.GetTokenAsync(
                new TokenRequestContext(["https://api.fabric.microsoft.com/.default"]),
                refreshToken);
        }
        catch (OperationCanceledException) when (!cancellationToken.IsCancellationRequested)
        {
            return;
        }
        using var request = new HttpRequestMessage(
            HttpMethod.Post,
            BuildRefreshMetadataUri());
        request.Headers.Authorization =
            new AuthenticationHeaderValue("Bearer", token.Token);
        request.Content = CreateMetadataRefreshContent(tables);
        HttpResponseMessage response;
        try
        {
            response = await httpClient.SendAsync(request, refreshToken);
        }
        catch (OperationCanceledException) when (!cancellationToken.IsCancellationRequested)
        {
            return;
        }

        using (response)
        {
            try
            {
                var body = await response.Content.ReadAsStringAsync(refreshToken);
                if (response.StatusCode == HttpStatusCode.Accepted)
                {
                    body = await TryReadMetadataOperationResultAsync(
                        response,
                        token.Token,
                        refreshToken);
                    if (string.IsNullOrWhiteSpace(body))
                    {
                        return;
                    }
                }
                else if (response.StatusCode is HttpStatusCode.Conflict or
                    HttpStatusCode.TooManyRequests)
                {
                    return;
                }
                else if (!response.IsSuccessStatusCode)
                {
                    throw new FabricQueryException(
                        "metadata_refresh_failed",
                        "Fabric could not refresh the SQL metadata for the required Dataverse tables.",
                        body);
                }

                var failure = CreateMetadataFailure(body, newlyCreatedTables);
                if (failure is not null)
                {
                    throw failure;
                }
            }
            catch (OperationCanceledException) when (!cancellationToken.IsCancellationRequested)
            {
                return;
            }
        }
    }

    internal static HttpContent CreateMetadataRefreshContent(
        IReadOnlyCollection<string> tables)
    {
        return JsonContent.Create(new
        {
            recreateTables = false,
            tables = new[]
            {
                new
                {
                    schema = "dataverse",
                    tableNames = tables
                }
            },
            timeout = new
            {
                timeUnit = "Seconds",
                value = 10
            }
        });
    }

    internal static HttpContent CreateShortcutContent(
        string name,
        DataverseShortcutTarget sourceTarget)
    {
        return JsonContent.Create(new
        {
            path = "Tables/dataverse",
            name,
            target = new
            {
                dataverse = new
                {
                    sourceTarget.ConnectionId,
                    sourceTarget.DeltaLakeFolder,
                    sourceTarget.EnvironmentDomain,
                    sourceTarget.TableName
                }
            }
        });
    }

    private async Task<string?> TryReadMetadataOperationResultAsync(
        HttpResponseMessage initialResponse,
        string accessToken,
        CancellationToken cancellationToken)
    {
        if (!initialResponse.Headers.TryGetValues(
                "x-ms-operation-id",
                out var operationIds))
        {
            return null;
        }

        var operationId = operationIds.FirstOrDefault();
        if (string.IsNullOrWhiteSpace(operationId))
        {
            return null;
        }

        var deadline = DateTimeOffset.UtcNow.AddSeconds(10);
        var retryAfter = initialResponse.Headers.RetryAfter?.Delta ??
            TimeSpan.FromSeconds(1);
        if (retryAfter < TimeSpan.FromSeconds(1))
        {
            retryAfter = TimeSpan.FromSeconds(1);
        }
        while (DateTimeOffset.UtcNow < deadline)
        {
            await Task.Delay(
                retryAfter > TimeSpan.FromSeconds(2)
                    ? TimeSpan.FromSeconds(2)
                    : retryAfter,
                cancellationToken);
            using var statusRequest = new HttpRequestMessage(
                HttpMethod.Get,
                new Uri(
                    $"https://api.fabric.microsoft.com/v1/operations/{Uri.EscapeDataString(operationId)}",
                    UriKind.Absolute));
            statusRequest.Headers.Authorization =
                new AuthenticationHeaderValue("Bearer", accessToken);
            using var statusResponse = await httpClient.SendAsync(
                statusRequest,
                cancellationToken);
            var statusBody = await statusResponse.Content.ReadAsStringAsync(
                cancellationToken);
            if (!statusResponse.IsSuccessStatusCode)
            {
                return null;
            }

            JsonDocument statusJson;
            try
            {
                statusJson = JsonDocument.Parse(statusBody);
            }
            catch (JsonException)
            {
                return null;
            }

            using (statusJson)
            {
            if (!statusJson.RootElement.TryGetProperty("status", out var status) ||
                status.ValueKind != JsonValueKind.String)
            {
                return null;
            }

            if (string.Equals(
                    status.GetString(),
                    "Failed",
                    StringComparison.OrdinalIgnoreCase))
            {
                throw new FabricQueryException(
                    "metadata_refresh_failed",
                    "Fabric could not complete the SQL metadata refresh.",
                    statusBody);
            }

            if (!string.Equals(
                    status.GetString(),
                    "Succeeded",
                    StringComparison.OrdinalIgnoreCase))
            {
                continue;
            }

            using var resultRequest = new HttpRequestMessage(
                HttpMethod.Get,
                new Uri(
                    $"https://api.fabric.microsoft.com/v1/operations/" +
                    $"{Uri.EscapeDataString(operationId)}/result",
                    UriKind.Absolute));
            resultRequest.Headers.Authorization =
                new AuthenticationHeaderValue("Bearer", accessToken);
            using var resultResponse = await httpClient.SendAsync(
                resultRequest,
                cancellationToken);
            return resultResponse.IsSuccessStatusCode
                ? await resultResponse.Content.ReadAsStringAsync(cancellationToken)
                : null;
            }
        }

        return null;
    }

    internal static FabricQueryException? CreateMetadataFailure(
        string responseBody,
        IReadOnlyCollection<string>? newlyCreatedTables = null)
    {
        if (string.IsNullOrWhiteSpace(responseBody))
        {
            return null;
        }

        JsonDocument json;
        try
        {
            json = JsonDocument.Parse(responseBody);
        }
        catch (JsonException)
        {
            return null;
        }

        using (json)
        {
        if (!json.RootElement.TryGetProperty("value", out var values) ||
            values.ValueKind != JsonValueKind.Array)
        {
            return null;
        }

        var failures = values.EnumerateArray()
            .Where(value =>
                value.TryGetProperty("status", out var status) &&
                status.ValueKind == JsonValueKind.String &&
                string.Equals(status.GetString(), "Failure", StringComparison.OrdinalIgnoreCase))
            .Select(value => new
            {
                Table = value.TryGetProperty("tableName", out var tableName) &&
                    tableName.ValueKind == JsonValueKind.String
                        ? tableName.GetString()
                        : null,
                Code = value.TryGetProperty("error", out var error) &&
                    error.TryGetProperty("errorCode", out var errorCode) &&
                    errorCode.ValueKind == JsonValueKind.String
                        ? errorCode.GetString()
                        : null,
                Message = value.TryGetProperty("error", out error) &&
                    error.TryGetProperty("message", out var message) &&
                    message.ValueKind == JsonValueKind.String
                        ? message.GetString()
                        : null
            })
            .ToList();
        if (failures.Count == 0)
        {
            return null;
        }

        var actionableFailures = failures
            .Where(failure =>
                !string.Equals(
                    failure.Code,
                    "DeltaTableNotFound",
                    StringComparison.OrdinalIgnoreCase) ||
                !(newlyCreatedTables ?? [])
                    .Contains(
                        failure.Table?.Split('.').LastOrDefault() ?? "",
                        StringComparer.OrdinalIgnoreCase))
            .ToList();
        if (actionableFailures.Count == 0)
        {
            return null;
        }

        var missingDeltaTables = actionableFailures
            .Where(failure =>
                string.Equals(
                    failure.Code,
                    "DeltaTableNotFound",
                    StringComparison.OrdinalIgnoreCase))
            .Select(failure => failure.Table)
            .Where(table => !string.IsNullOrWhiteSpace(table))
            .Distinct(StringComparer.OrdinalIgnoreCase)
            .ToList();
        if (missingDeltaTables.Count > 0)
        {
            return new FabricQueryException(
                "dependency_source_not_ready",
                "Fabric created the shortcuts, but Dataverse Managed Lake does not contain " +
                "queryable Delta tables for: " + string.Join(", ", missingDeltaTables) + ". " +
                "Their initial Link to Microsoft Fabric synchronization must finish before " +
                "they can be queried and can take up to 60 minutes. In Power Apps, open " +
                "Link to Microsoft Fabric > Manage tables and select Refresh Fabric links/tables, " +
                "then retry the segment preview.");
        }

        var authenticationFailures = actionableFailures
            .Where(failure =>
                failure.Code?.IndexOf("Adal", StringComparison.OrdinalIgnoreCase) >= 0 ||
                failure.Message?.IndexOf(
                    "Microsoft Entra ID token",
                    StringComparison.OrdinalIgnoreCase) >= 0)
            .Select(failure => failure.Table)
            .Where(table => !string.IsNullOrWhiteSpace(table))
            .Distinct(StringComparer.OrdinalIgnoreCase)
            .ToList();
        if (authenticationFailures.Count > 0)
        {
            return new FabricQueryException(
                "dependency_connection_invalid",
                "Fabric could not read the Dataverse shortcut targets because the cloud " +
                "connection credentials are no longer valid. Update the Dataverse connection " +
                "under Manage connections and gateways, then retry. Tables: " +
                string.Join(", ", authenticationFailures) + ".");
        }

        return new FabricQueryException(
            "metadata_refresh_failed",
            "Fabric could not synchronize SQL metadata for: " +
            string.Join(
                ", ",
                actionableFailures.Select(failure => failure.Table)
                    .Where(table => !string.IsNullOrWhiteSpace(table))) + ".",
            responseBody);
        }
    }

    private async Task<Dictionary<string, DataverseShortcutTarget?>>
        ListExistingTargetsAsync(
        string accessToken,
        CancellationToken cancellationToken)
    {
        var existing = new Dictionary<string, DataverseShortcutTarget?>(
            StringComparer.OrdinalIgnoreCase);
        string? continuationToken = null;
        do
        {
            using var requestMessage = new HttpRequestMessage(
                HttpMethod.Get,
                BuildListShortcutUri(continuationToken));
            requestMessage.Headers.Authorization =
                new AuthenticationHeaderValue("Bearer", accessToken);
            using var listResponse = await httpClient.SendAsync(
                requestMessage,
                cancellationToken);
            var listBody = await listResponse.Content.ReadAsStringAsync(
                cancellationToken);
            if (!listResponse.IsSuccessStatusCode)
            {
                throw new FabricQueryException(
                    "shortcut_list_failed",
                    "The existing Fabric shortcuts could not be read.",
                    listBody);
            }

            using var listJson = JsonDocument.Parse(listBody);
            if (listJson.RootElement.TryGetProperty("value", out var values))
            {
                foreach (var shortcut in values.EnumerateArray())
                {
                    if (shortcut.TryGetProperty("name", out var name) &&
                        name.ValueKind == JsonValueKind.String)
                    {
                        var shortcutName = name.GetString()!;
                        existing[shortcutName] =
                            TryReadSourceTarget(shortcut, out _, out var target)
                                ? target
                                : null;
                    }
                }
            }

            continuationToken =
                listJson.RootElement.TryGetProperty(
                    "continuationToken",
                    out var continuation) &&
                continuation.ValueKind == JsonValueKind.String
                    ? continuation.GetString()
                    : null;
        }
        while (!string.IsNullOrEmpty(continuationToken));
        return existing;
    }

    internal static bool TargetsMatch(
        DataverseShortcutTarget left,
        DataverseShortcutTarget right)
    {
        return string.Equals(
                left.ConnectionId,
                right.ConnectionId,
                StringComparison.OrdinalIgnoreCase) &&
            string.Equals(
                left.DeltaLakeFolder,
                right.DeltaLakeFolder,
                StringComparison.OrdinalIgnoreCase) &&
            string.Equals(
                left.EnvironmentDomain.TrimEnd('/'),
                right.EnvironmentDomain.TrimEnd('/'),
                StringComparison.OrdinalIgnoreCase) &&
            string.Equals(
                left.TableName,
                right.TableName,
                StringComparison.OrdinalIgnoreCase);
    }

    private async Task<Dictionary<string, DataverseShortcutTarget>> ListSourceTargetsAsync(
        string accessToken,
        CancellationToken cancellationToken)
    {
        var targets = new Dictionary<string, DataverseShortcutTarget>(
            StringComparer.OrdinalIgnoreCase);
        string? continuationToken = null;
        do
        {
            using var request = new HttpRequestMessage(
                HttpMethod.Get,
                BuildListSourceShortcutUri(continuationToken));
            request.Headers.Authorization =
                new AuthenticationHeaderValue("Bearer", accessToken);
            using var response = await httpClient.SendAsync(request, cancellationToken);
            var body = await response.Content.ReadAsStringAsync(cancellationToken);
            if (!response.IsSuccessStatusCode)
            {
                throw new FabricQueryException(
                    "shortcut_source_list_failed",
                    "The Dataverse Managed Lake shortcuts could not be read.",
                    body);
            }

            using var json = JsonDocument.Parse(body);
            if (json.RootElement.TryGetProperty("value", out var values))
            {
                foreach (var shortcut in values.EnumerateArray())
                {
                    if (!IsRootSourceShortcut(shortcut) ||
                        !TryReadSourceTarget(shortcut, out var name, out var target))
                    {
                        continue;
                    }

                    targets[name] = target;
                }
            }

            continuationToken =
                json.RootElement.TryGetProperty(
                    "continuationToken",
                    out var continuation) &&
                continuation.ValueKind == JsonValueKind.String
                    ? continuation.GetString()
                    : null;
        }
        while (!string.IsNullOrEmpty(continuationToken));
        return targets;
    }

    internal static bool IsRootSourceShortcut(JsonElement shortcut)
    {
        return shortcut.TryGetProperty("path", out var path) &&
            path.ValueKind == JsonValueKind.String &&
            string.Equals(
                path.GetString()?.Trim('/'),
                "Tables",
                StringComparison.OrdinalIgnoreCase);
    }

    internal static bool TryReadSourceTarget(
        JsonElement shortcut,
        out string name,
        out DataverseShortcutTarget target)
    {
        name = "";
        target = default!;
        if (!shortcut.TryGetProperty("name", out var nameElement) ||
            nameElement.ValueKind != JsonValueKind.String ||
            !shortcut.TryGetProperty("target", out var targetElement) ||
            !targetElement.TryGetProperty("type", out var typeElement) ||
            typeElement.ValueKind != JsonValueKind.String ||
            !string.Equals(
                typeElement.GetString(),
                "Dataverse",
                StringComparison.OrdinalIgnoreCase) ||
            !targetElement.TryGetProperty("dataverse", out var dataverse))
        {
            return false;
        }

        name = nameElement.GetString()!;
        var connectionId = ReadJsonString(dataverse, "connectionId");
        var deltaLakeFolder = ReadJsonString(dataverse, "deltaLakeFolder");
        var environmentDomain = ReadJsonString(dataverse, "environmentDomain");
        var tableName = ReadJsonString(dataverse, "tableName");
        if (string.IsNullOrWhiteSpace(connectionId) ||
            string.IsNullOrWhiteSpace(deltaLakeFolder) ||
            string.IsNullOrWhiteSpace(environmentDomain) ||
            string.IsNullOrWhiteSpace(tableName))
        {
            return false;
        }

        target = new DataverseShortcutTarget(
            connectionId,
            deltaLakeFolder,
            environmentDomain,
            tableName);
        return true;
    }

    private static string? ReadJsonString(JsonElement value, string propertyName)
    {
        return value.TryGetProperty(propertyName, out var property) &&
            property.ValueKind == JsonValueKind.String
                ? property.GetString()
                : null;
    }

    private Uri BuildListShortcutUri(string? continuationToken)
    {
        var uri =
            $"https://api.fabric.microsoft.com/v1/workspaces/{Uri.EscapeDataString(workspaceId)}" +
            $"/items/{Uri.EscapeDataString(lakehouseId)}/shortcuts" +
            "?parentPath=Tables%2Fdataverse";
        if (!string.IsNullOrEmpty(continuationToken))
        {
            uri += "&continuationToken=" + Uri.EscapeDataString(continuationToken);
        }

        return new Uri(uri, UriKind.Absolute);
    }

    private Uri BuildCreateShortcutUri()
    {
        return new Uri(
            $"https://api.fabric.microsoft.com/v1/workspaces/{Uri.EscapeDataString(workspaceId)}" +
            $"/items/{Uri.EscapeDataString(lakehouseId)}/shortcuts" +
            "?shortcutConflictPolicy=CreateOrOverwrite",
            UriKind.Absolute);
    }

    private Uri BuildListSourceShortcutUri(string? continuationToken)
    {
        var uri =
            $"https://api.fabric.microsoft.com/v1/workspaces/{Uri.EscapeDataString(workspaceId)}" +
            $"/items/{Uri.EscapeDataString(dataverseLakehouseId)}/shortcuts" +
            "?parentPath=Tables";
        if (!string.IsNullOrEmpty(continuationToken))
        {
            uri += "&continuationToken=" + Uri.EscapeDataString(continuationToken);
        }

        return new Uri(uri, UriKind.Absolute);
    }

    private Uri BuildRefreshMetadataUri()
    {
        return new Uri(
            $"https://api.fabric.microsoft.com/v1/workspaces/{Uri.EscapeDataString(workspaceId)}" +
            $"/sqlEndpoints/{Uri.EscapeDataString(sqlEndpointId)}/refreshMetadata",
            UriKind.Absolute);
    }

    private static string ReadRequiredSetting(string name)
    {
        var value = Environment.GetEnvironmentVariable(name)?.Trim();
        return !string.IsNullOrEmpty(value)
            ? value
            : throw new InvalidOperationException(
                $"The app setting '{name}' is not configured.");
    }

    internal sealed record DataverseShortcutTarget(
        string ConnectionId,
        string DeltaLakeFolder,
        string EnvironmentDomain,
        string TableName);
}
