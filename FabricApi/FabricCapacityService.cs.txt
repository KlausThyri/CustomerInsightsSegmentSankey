#if FABRIC_API
using System.Net;
using System.Net.Http.Headers;
using System.Text.Json;
using System.Text.RegularExpressions;
using Azure.Core;

namespace CustomerInsightsSegmentSankey.FabricApi;

internal sealed class FabricCapacityService : IFabricCapacityReadiness
{
    private const string ArmScope = "https://management.azure.com/.default";
    private const string ApiVersion = "2023-11-01";
    private static readonly SemaphoreSlim ResumeLock = new(1, 1);
    private static readonly Regex CapacityResourceIdPattern = new(
        "^/subscriptions/[0-9a-f-]{36}/resourceGroups/[^/]+/providers/Microsoft\\.Fabric/capacities/[^/]+$",
        RegexOptions.IgnoreCase | RegexOptions.CultureInvariant);

    private readonly HttpClient httpClient;
    private readonly TokenCredential credential;
    private readonly FabricCapacityReadinessCache readinessCache;
    private readonly string capacityResourceId;
    private readonly Func<TimeSpan, CancellationToken, Task> delay;

    /// <summary>
    /// Production constructor. <paramref name="credential"/> and
    /// <paramref name="readinessCache"/> are resolved from DI as singletons
    /// (see <c>Program.cs</c>) so the Fabric REST/ARM token cache inside
    /// <paramref name="credential"/> and the short-TTL active-capacity
    /// readiness cache are both shared across every transient
    /// <see cref="FabricCapacityService"/> instance, including the one used
    /// by <see cref="FabricSqlWarmupService"/> startup warm-up.
    /// </summary>
    public FabricCapacityService(
        HttpClient httpClient,
        TokenCredential credential,
        FabricCapacityReadinessCache readinessCache)
        : this(
            httpClient,
            credential,
            readinessCache,
            ReadRequiredSetting("FABRIC_CAPACITY_RESOURCE_ID"),
            static (duration, cancellationToken) => Task.Delay(duration, cancellationToken))
    {
    }

    internal FabricCapacityService(
        HttpClient httpClient,
        TokenCredential credential,
        FabricCapacityReadinessCache readinessCache,
        string capacityResourceId,
        Func<TimeSpan, CancellationToken, Task> delay)
    {
        this.httpClient = httpClient;
        this.credential = credential;
        this.readinessCache = readinessCache;
        this.capacityResourceId = NormalizeResourceId(capacityResourceId);
        this.delay = delay;
    }

    public async Task<bool> EnsureActiveAsync(CancellationToken cancellationToken)
    {
        if (readinessCache.IsRecentlyConfirmedActive(capacityResourceId))
        {
            return false;
        }

        var token = await credential.GetTokenAsync(
            new TokenRequestContext([ArmScope]),
            cancellationToken);
        var state = await ReadStateAsync(token.Token, cancellationToken);
        if (IsActive(state))
        {
            // Already active on the very first check this call made: no
            // resume/wait was necessary, so this is not a transition.
            readinessCache.MarkConfirmedActive(capacityResourceId);
            return false;
        }

        await ResumeLock.WaitAsync(cancellationToken);
        try
        {
            state = await ReadStateAsync(token.Token, cancellationToken);
            if (IsActive(state))
            {
                // The initial check (above) observed the capacity as not
                // active, and it is active now - e.g. a concurrent caller
                // resumed it while this call waited for the lock. From this
                // call's perspective that is a transition a concurrently
                // started SQL evaluation may need to retry against.
                readinessCache.MarkConfirmedActive(capacityResourceId);
                return true;
            }

            if (string.Equals(state, "Paused", StringComparison.OrdinalIgnoreCase))
            {
                await ResumeAsync(token.Token, cancellationToken);
            }
            else if (!IsStarting(state))
            {
                throw new FabricQueryException(
                    "capacity_unavailable",
                    $"The Fabric capacity is in state '{state ?? "Unknown"}' and cannot be used. " +
                    "Open the Azure portal, verify the capacity, and retry the segment preview.");
            }

            for (var attempt = 0; attempt < 6; attempt++)
            {
                await delay(TimeSpan.FromSeconds(2), cancellationToken);
                state = await ReadStateAsync(token.Token, cancellationToken);
                if (IsActive(state))
                {
                    readinessCache.MarkConfirmedActive(capacityResourceId);
                    return true;
                }
                if (!IsStarting(state) && !string.Equals(
                        state,
                        "Paused",
                        StringComparison.OrdinalIgnoreCase))
                {
                    throw new FabricQueryException(
                        "capacity_unavailable",
                        $"The Fabric capacity entered state '{state ?? "Unknown"}' while starting. " +
                        "Open the Azure portal, verify the capacity, and retry the segment preview.");
                }
            }
        }
        finally
        {
            ResumeLock.Release();
        }

        throw new FabricQueryException(
            "capacity_starting",
            "The Fabric capacity is starting automatically. Retry the segment preview in a few seconds.");
    }

    private async Task<string?> ReadStateAsync(
        string accessToken,
        CancellationToken cancellationToken)
    {
        using var request = CreateRequest(HttpMethod.Get, capacityResourceId, accessToken);
        using var response = await httpClient.SendAsync(request, cancellationToken);
        if (response.StatusCode == HttpStatusCode.Forbidden)
        {
            throw new FabricQueryException(
                "capacity_permission_missing",
                "The Segment Preview managed identity cannot read or start the Fabric capacity. " +
                "Run Setup Center > Install everything to grant access.");
        }
        if (!response.IsSuccessStatusCode)
        {
            throw await CreateRequestFailureAsync(
                "capacity_status_failed",
                "The Fabric capacity status could not be read.",
                response,
                cancellationToken);
        }

        using var document = JsonDocument.Parse(
            await response.Content.ReadAsStreamAsync(cancellationToken));
        return document.RootElement.TryGetProperty("properties", out var properties) &&
               properties.TryGetProperty("state", out var state)
            ? state.GetString()
            : null;
    }

    private async Task ResumeAsync(
        string accessToken,
        CancellationToken cancellationToken)
    {
        using var request = CreateRequest(
            HttpMethod.Post,
            capacityResourceId + "/resume",
            accessToken);
        using var response = await httpClient.SendAsync(request, cancellationToken);
        if (response.StatusCode == HttpStatusCode.Forbidden)
        {
            throw new FabricQueryException(
                "capacity_permission_missing",
                "The Segment Preview managed identity cannot start the Fabric capacity. " +
                "Run Setup Center > Install everything to grant access.");
        }
        if (response.StatusCode is not (HttpStatusCode.OK or HttpStatusCode.Accepted))
        {
            throw await CreateRequestFailureAsync(
                "capacity_resume_failed",
                "Azure rejected the automatic Fabric capacity start.",
                response,
                cancellationToken);
        }
    }

    private static HttpRequestMessage CreateRequest(
        HttpMethod method,
        string resourcePath,
        string accessToken)
    {
        var separator = resourcePath.IndexOf('?') >= 0 ? '&' : '?';
        var request = new HttpRequestMessage(
            method,
            "https://management.azure.com" + resourcePath + separator +
            "api-version=" + ApiVersion);
        request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", accessToken);
        return request;
    }

    private static async Task<FabricQueryException> CreateRequestFailureAsync(
        string code,
        string message,
        HttpResponseMessage response,
        CancellationToken cancellationToken)
    {
        var detail = await response.Content.ReadAsStringAsync(cancellationToken);
        return new FabricQueryException(
            code,
            message,
            $"Azure returned {(int)response.StatusCode} {response.ReasonPhrase}. {detail}".Trim());
    }

    private static bool IsActive(string? state) =>
        string.Equals(state, "Active", StringComparison.OrdinalIgnoreCase);

    private static bool IsStarting(string? state) =>
        string.Equals(state, "Provisioning", StringComparison.OrdinalIgnoreCase) ||
        string.Equals(state, "Updating", StringComparison.OrdinalIgnoreCase) ||
        string.Equals(state, "Resuming", StringComparison.OrdinalIgnoreCase);

    private static string NormalizeResourceId(string value)
    {
        var normalized = (value ?? string.Empty).Trim().TrimEnd('/');
        if (!CapacityResourceIdPattern.IsMatch(normalized))
        {
            throw new InvalidOperationException(
                "FABRIC_CAPACITY_RESOURCE_ID must be the Azure resource ID of a Microsoft.Fabric capacity.");
        }
        return normalized;
    }

    private static string ReadRequiredSetting(string name)
    {
        var value = Environment.GetEnvironmentVariable(name);
        if (string.IsNullOrWhiteSpace(value))
        {
            throw new InvalidOperationException($"The required app setting '{name}' is missing.");
        }
        return value;
    }
}
#endif
