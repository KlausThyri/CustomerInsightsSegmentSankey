#if FABRIC_API
using System.Reflection;

namespace CustomerInsightsSegmentSankey.FabricApi;

internal static class SegmentDiagnosticsFactory
{
    public static SegmentPerformanceDiagnostics Create(
        SegmentCountRequest request,
        string requestId)
    {
        var diagnostics = new SegmentPerformanceDiagnostics
        {
            ApiVersion = Assembly.GetExecutingAssembly()
                .GetName()
                .Version?
                .ToString() ?? string.Empty,
            RequestId = requestId,
            QueryComplexity = Analyze(request.Query)
        };
        diagnostics.Runtime.BusinessUnitScoping = HasBusinessUnit(request.Query);
        PopulateIdentifiers(diagnostics.Identifiers);
        return diagnostics;
    }

    internal static SegmentQueryComplexity Analyze(SegmentQueryNode? root)
    {
        var result = new SegmentQueryComplexity();
        VisitQuery(root, 1, result);
        return result;
    }

    private static void VisitQuery(
        SegmentQueryNode? query,
        int depth,
        SegmentQueryComplexity result)
    {
        if (query is null || depth > 64)
        {
            return;
        }

        result.MaximumDepth = Math.Max(result.MaximumDepth, depth);
        VisitOperand(query.FirstOperand, depth, result);
        foreach (var operation in query.SetOperations?.Take(1000) ?? [])
        {
            if (string.Equals(operation.Operator, "union", StringComparison.OrdinalIgnoreCase))
            {
                result.Unions++;
            }
            else if (string.Equals(operation.Operator, "except", StringComparison.OrdinalIgnoreCase))
            {
                result.Exclusions++;
            }
            VisitOperand(operation.Operand, depth, result);
        }
    }

    private static void VisitOperand(
        SegmentOperand? operand,
        int depth,
        SegmentQueryComplexity result)
    {
        if (operand is null)
        {
            return;
        }

        if (string.Equals(operand.Kind, "interaction", StringComparison.OrdinalIgnoreCase))
        {
            result.InteractionFilters++;
        }
        foreach (var step in operand.Steps?.Take(1000) ?? [])
        {
            if (string.Equals(step.Kind, "relationship", StringComparison.OrdinalIgnoreCase))
            {
                result.RelationshipFilters++;
            }
            else
            {
                result.ProfileFilters++;
            }
            CountConsent(step.Condition, 1, result);
        }
        CountConsent(operand.Filter, 1, result);
        VisitQuery(operand.Query, depth + 1, result);
    }

    private static void CountConsent(
        SegmentCondition? condition,
        int depth,
        SegmentQueryComplexity result)
    {
        if (condition is null || depth > 64)
        {
            return;
        }
        if (string.Equals(condition.Kind, "consent", StringComparison.OrdinalIgnoreCase))
        {
            result.ConsentFilters++;
        }
        foreach (var child in condition.Children?.Take(1000) ?? [])
        {
            CountConsent(child, depth + 1, result);
        }
    }

    private static bool HasBusinessUnit(SegmentQueryNode? query)
    {
        if (query is null)
        {
            return false;
        }
        if (query.BusinessUnitId.HasValue)
        {
            return true;
        }
        if (HasBusinessUnit(query.FirstOperand?.Query))
        {
            return true;
        }
        return query.SetOperations?.Any(
            operation => HasBusinessUnit(operation.Operand?.Query)) == true;
    }

    private static void PopulateIdentifiers(SegmentDiagnosticIdentifiers identifiers)
    {
        var capacityId = Environment.GetEnvironmentVariable("FABRIC_CAPACITY_RESOURCE_ID");
        if (!string.IsNullOrWhiteSpace(capacityId))
        {
            var parts = capacityId.Split('/', StringSplitOptions.RemoveEmptyEntries);
            if (parts.Length >= 4)
            {
                identifiers.SubscriptionId = parts[1];
                identifiers.ResourceGroup = parts[3];
            }
        }
        identifiers.WorkspaceId =
            Environment.GetEnvironmentVariable("FABRIC_WORKSPACE_ID");
        identifiers.LakehouseId =
            Environment.GetEnvironmentVariable("FABRIC_SERVING_LAKEHOUSE_ID");
    }
}
#endif
