#if FABRIC_API
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;

namespace CustomerInsightsSegmentSankey.FabricApi;

/// <summary>
/// Issues and validates the opaque, authenticated + encrypted tokens used by
/// <c>/api/segment-counts</c> (queryToken) and <c>/api/segment-members</c>
/// (continuationToken). Both token families are AES-GCM (AEAD) protected
/// using keys derived from the existing <c>BEHAVIORAL_API_KEY</c> secret via
/// HKDF-SHA256 with distinct, purpose-scoped info strings so the two token
/// families cannot be swapped for one another.
/// </summary>
internal sealed class QueryTokenService
{
    private const string QueryTokenInfo = "CustomerInsightsSegmentSankey.segment-query-token.v1";
    private const string ContinuationTokenInfo = "CustomerInsightsSegmentSankey.segment-continuation-token.v1";
    private const int NonceSizeInBytes = 12;
    private const int TagSizeInBytes = 16;

    private static readonly TimeSpan QueryTokenLifetime = TimeSpan.FromMinutes(30);
    private static readonly JsonSerializerOptions SerializerOptions = new(JsonSerializerDefaults.Web);

    private readonly byte[] queryTokenKey;
    private readonly byte[] continuationTokenKey;

    public QueryTokenService()
    {
        var secret = RequiredSetting("BEHAVIORAL_API_KEY");
        queryTokenKey = DeriveKey(secret, QueryTokenInfo);
        continuationTokenKey = DeriveKey(secret, ContinuationTokenInfo);
    }

    public string IssueQueryToken(SegmentQueryNode query)
    {
        return IssueQueryToken(query, DateTimeOffset.UtcNow.Add(QueryTokenLifetime));
    }

    internal string IssueQueryToken(SegmentQueryNode query, DateTimeOffset expiresAt)
    {
        var payload = new QueryTokenPayload
        {
            Query = query,
            ExpiresAt = expiresAt
        };
        return Protect(queryTokenKey, JsonSerializer.Serialize(payload, SerializerOptions));
    }

    public QueryTokenPayload ValidateQueryToken(string? token)
    {
        if (string.IsNullOrWhiteSpace(token))
        {
            throw new SegmentQueryException(
                "invalid_query_token",
                "queryToken is required.");
        }

        var json = Unprotect(queryTokenKey, token, "invalid_query_token");
        QueryTokenPayload? payload;
        try
        {
            payload = JsonSerializer.Deserialize<QueryTokenPayload>(json, SerializerOptions);
        }
        catch (JsonException)
        {
            throw new SegmentQueryException(
                "invalid_query_token",
                "The query token payload could not be parsed.");
        }

        if (payload?.Query?.FirstOperand is null)
        {
            throw new SegmentQueryException(
                "invalid_query_token",
                "The query token does not contain a valid query.");
        }

        if (DateTimeOffset.UtcNow > payload.ExpiresAt)
        {
            throw new SegmentQueryException(
                "query_token_expired",
                "The query token has expired. Request a new segment count to continue.");
        }

        return payload;
    }

    public string HashQueryToken(string queryToken)
    {
        return Convert.ToBase64String(SHA256.HashData(Encoding.UTF8.GetBytes(queryToken)));
    }

    public string IssueContinuationToken(ContinuationTokenPayload payload)
    {
        return Protect(continuationTokenKey, JsonSerializer.Serialize(payload, SerializerOptions));
    }

    public ContinuationTokenPayload ValidateContinuationToken(string token)
    {
        var json = Unprotect(continuationTokenKey, token, "invalid_continuation_token");
        ContinuationTokenPayload? payload;
        try
        {
            payload = JsonSerializer.Deserialize<ContinuationTokenPayload>(json, SerializerOptions);
        }
        catch (JsonException)
        {
            throw new SegmentQueryException(
                "invalid_continuation_token",
                "The continuation token payload could not be parsed.");
        }

        if (payload is null || string.IsNullOrEmpty(payload.QueryTokenHash))
        {
            throw new SegmentQueryException(
                "invalid_continuation_token",
                "The continuation token is malformed.");
        }

        if (DateTimeOffset.UtcNow > payload.ExpiresAt)
        {
            throw new SegmentQueryException(
                "continuation_token_expired",
                "The continuation token has expired. Request a new segment count to continue.");
        }

        return payload;
    }

    private static byte[] DeriveKey(string secret, string info)
    {
        return HKDF.DeriveKey(
            HashAlgorithmName.SHA256,
            Encoding.UTF8.GetBytes(secret),
            outputLength: 32,
            salt: null,
            info: Encoding.UTF8.GetBytes(info));
    }

    private static string Protect(byte[] key, string plaintext)
    {
        var plaintextBytes = Encoding.UTF8.GetBytes(plaintext);
        var nonce = RandomNumberGenerator.GetBytes(NonceSizeInBytes);
        var ciphertext = new byte[plaintextBytes.Length];
        var tag = new byte[TagSizeInBytes];
        using (var aes = new AesGcm(key, TagSizeInBytes))
        {
            aes.Encrypt(nonce, plaintextBytes, ciphertext, tag);
        }

        var result = new byte[nonce.Length + ciphertext.Length + tag.Length];
        Buffer.BlockCopy(nonce, 0, result, 0, nonce.Length);
        Buffer.BlockCopy(ciphertext, 0, result, nonce.Length, ciphertext.Length);
        Buffer.BlockCopy(tag, 0, result, nonce.Length + ciphertext.Length, tag.Length);
        return Base64UrlEncode(result);
    }

    private static string Unprotect(byte[] key, string token, string errorCode)
    {
        byte[] raw;
        try
        {
            raw = Base64UrlDecode(token);
        }
        catch (FormatException)
        {
            throw new SegmentQueryException(errorCode, "The token is not well-formed.");
        }

        if (raw.Length < NonceSizeInBytes + TagSizeInBytes)
        {
            throw new SegmentQueryException(errorCode, "The token is too short to be valid.");
        }

        var nonce = raw.AsSpan(0, NonceSizeInBytes);
        var ciphertext = raw.AsSpan(NonceSizeInBytes, raw.Length - NonceSizeInBytes - TagSizeInBytes);
        var tag = raw.AsSpan(raw.Length - TagSizeInBytes, TagSizeInBytes);
        var plaintext = new byte[ciphertext.Length];
        try
        {
            using var aes = new AesGcm(key, TagSizeInBytes);
            aes.Decrypt(nonce, ciphertext, tag, plaintext);
        }
        catch (CryptographicException)
        {
            throw new SegmentQueryException(
                errorCode,
                "The token failed authentication and may have been tampered with.");
        }

        return Encoding.UTF8.GetString(plaintext);
    }

    private static string Base64UrlEncode(byte[] data)
    {
        return Convert.ToBase64String(data)
            .TrimEnd('=')
            .Replace('+', '-')
            .Replace('/', '_');
    }

    private static byte[] Base64UrlDecode(string value)
    {
        var padded = value.Replace('-', '+').Replace('_', '/');
        switch (padded.Length % 4)
        {
            case 2:
                padded += "==";
                break;
            case 3:
                padded += "=";
                break;
            case 1:
                throw new FormatException("Invalid base64url string length.");
        }

        var decoded = Convert.FromBase64String(padded);
        if (!string.Equals(Base64UrlEncode(decoded), value, StringComparison.Ordinal))
        {
            throw new FormatException("The base64url token is not canonical.");
        }

        return decoded;
    }

    private static string RequiredSetting(string name)
    {
        var value = Environment.GetEnvironmentVariable(name);
        if (string.IsNullOrWhiteSpace(value))
        {
            throw new InvalidOperationException(
                $"The application setting '{name}' is not configured.");
        }

        return value.Trim();
    }
}

internal sealed class QueryTokenPayload
{
    public SegmentQueryNode Query { get; set; } = new();

    public DateTimeOffset ExpiresAt { get; set; }
}

internal sealed class ContinuationTokenPayload
{
    public string QueryTokenHash { get; set; } = string.Empty;

    public int StageOrder { get; set; }

    public string ViewMode { get; set; } = string.Empty;

    public string? Search { get; set; }

    public string? FilterField { get; set; }

    public string? FilterOperator { get; set; }

    public string? FilterValue { get; set; }

    public string SortField { get; set; } = string.Empty;

    public string SortDirection { get; set; } = string.Empty;

    public bool LastValueIsNull { get; set; }

    public string? LastValue { get; set; }

    public Guid LastProfileId { get; set; }

    public DateTimeOffset ExpiresAt { get; set; }
}
#endif
