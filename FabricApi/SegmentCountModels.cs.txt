#if FABRIC_API
using System.Text.Json;
using System.Text.Json.Serialization;

namespace CustomerInsightsSegmentSankey.FabricApi;

public sealed class SegmentCountRequest
{
    public SegmentQueryNode Query { get; set; } = new();

    /// <summary>
    /// The Dataverse tables the caller has already resolved as required to
    /// evaluate <see cref="Query"/>. Scopes safe, optimistic dependency
    /// reconciliation to exactly these tables instead of discovering the
    /// full dependency set server-side.
    /// </summary>
    public List<string> RequiredDataverseTables { get; set; } = [];

    [JsonIgnore]
    internal SegmentPerformanceDiagnostics Diagnostics { get; set; } = new();
}

public sealed class SegmentQueryNode
{
    public Guid? BusinessUnitId { get; set; }

    public SegmentOperand? FirstOperand { get; set; }

    public List<SegmentSetOperation> SetOperations { get; set; } = [];
}

public sealed class SegmentSetOperation
{
    public string Operator { get; set; } = string.Empty;

    public SegmentOperand? Operand { get; set; }

    public string Label { get; set; } = string.Empty;

    public string Detail { get; set; } = string.Empty;
}

public sealed class SegmentOperand
{
    public string Kind { get; set; } = string.Empty;

    public string ProfileEntity { get; set; } = string.Empty;

    public string BaseLabel { get; set; } = string.Empty;

    public string BaseDetail { get; set; } = string.Empty;

    public List<SegmentFilterStep> Steps { get; set; } = [];

    public string EventLogicalName { get; set; } = string.Empty;

    public string EntityIdField { get; set; } = string.Empty;

    public SegmentCondition? Filter { get; set; }

    public SegmentHaving Having { get; set; } = new();

    public List<Guid> ProfileIds { get; set; } = [];

    public SegmentQueryNode? Query { get; set; }
}

public sealed class SegmentFilterStep
{
    public string Kind { get; set; } = string.Empty;

    public string Label { get; set; } = string.Empty;

    public string Detail { get; set; } = string.Empty;

    public SegmentCondition? Condition { get; set; }

    public string RelatedEntity { get; set; } = string.Empty;

    public string ProfileAttribute { get; set; } = string.Empty;

    public string RelatedAttribute { get; set; } = string.Empty;

    public bool IsOptional { get; set; }

    public List<SegmentRelationshipHop> Relationships { get; set; } = [];
}

public sealed class SegmentRelationshipHop
{
    public string Alias { get; set; } = string.Empty;

    public string RelatedEntity { get; set; } = string.Empty;

    public string SourceAttribute { get; set; } = string.Empty;

    public string RelatedAttribute { get; set; } = string.Empty;

    public bool IsOptional { get; set; }

    public string IntersectEntity { get; set; } = string.Empty;

    public string SourceIntersectAttribute { get; set; } = string.Empty;

    public string RelatedIntersectAttribute { get; set; } = string.Empty;

    public string SourcePrimaryAttribute { get; set; } = string.Empty;

    public string RelatedPrimaryAttribute { get; set; } = string.Empty;
}

public sealed class SegmentCondition
{
    public string Kind { get; set; } = string.Empty;

    public string? Field { get; set; }

    public string? Qualifier { get; set; }

    public string? Operator { get; set; }

    public JsonElement? Value { get; set; }

    public List<JsonElement>? Values { get; set; }

    public List<SegmentCondition>? Children { get; set; }

    public string? ProfileEmailField { get; set; }

    public Guid? PurposeId { get; set; }

    public Guid? TopicId { get; set; }

    public string? Channel { get; set; }
}

public sealed class SegmentHaving
{
    public string Metric { get; set; } = "Count";

    public string Operator { get; set; } = ">=";

    public long Threshold { get; set; } = 1;

    public string? WindowFunction { get; set; }

    public int? WindowValue { get; set; }
}

public sealed class SegmentCountResponse
{
    public List<SegmentCountStage> Stages { get; set; } = [];

    public List<SegmentSourceTable> SourceTables { get; set; } = [];

    public DateTimeOffset GeneratedAt { get; set; }

    /// <summary>
    /// An opaque, authenticated and encrypted token binding the validated
    /// <see cref="SegmentQueryNode"/> and an expiration to a follow-up call to
    /// <c>/api/segment-members</c>. The token never exposes stage SQL or PII.
    /// </summary>
    public string QueryToken { get; set; } = string.Empty;

    /// <summary>
    /// Dataverse tables that were newly added to Fabric while reconciling a
    /// dependency that was absent from the catalog. Empty on the common
    /// optimistic path, where every table in
    /// <see cref="SegmentCountRequest.RequiredDataverseTables"/> was already
    /// present.
    /// </summary>
    public List<string> AddedTables { get; set; } = [];

    /// <summary>
    /// True once the Fabric catalog is confirmed ready for every table in
    /// <see cref="SegmentCountRequest.RequiredDataverseTables"/>. Always true
    /// on a successful response; present so older callers can detect a
    /// stale/incompatible deployment if it is ever false.
    /// </summary>
    public bool CatalogReady { get; set; }

    public SegmentPerformanceDiagnostics Diagnostics { get; set; } = new();
}

public sealed class SegmentPerformanceDiagnostics
{
    public string SchemaVersion { get; set; } = "1.0";

    public DateTimeOffset CapturedAtUtc { get; set; } = DateTimeOffset.UtcNow;

    public string ApiVersion { get; set; } = string.Empty;

    public string RequestId { get; set; } = string.Empty;

    public string? ErrorCode { get; set; }

    public SegmentDiagnosticTimings TimingsMs { get; set; } = new();

    public SegmentQueryComplexity QueryComplexity { get; set; } = new();

    public SegmentDiagnosticRuntime Runtime { get; set; } = new();

    public List<string> SourceTables { get; set; } = [];

    public SegmentDiagnosticIdentifiers Identifiers { get; set; } = new();
}

public sealed class SegmentDiagnosticTimings
{
    public double ApiTotal { get; set; }

    public double CapacityReadiness { get; set; }

    public double Catalog { get; set; }

    public double SqlConnection { get; set; }

    public double SqlExecution { get; set; }
}

public sealed class SegmentQueryComplexity
{
    public int Stages { get; set; }

    public int ProfileFilters { get; set; }

    public int RelationshipFilters { get; set; }

    public int ConsentFilters { get; set; }

    public int InteractionFilters { get; set; }

    public int Unions { get; set; }

    public int Exclusions { get; set; }

    public int MaximumDepth { get; set; }
}

public sealed class SegmentDiagnosticRuntime
{
    public string Cache { get; set; } = "not_reported";

    public bool Coalesced { get; set; }

    public string CapacityState { get; set; } = "Unknown";

    public bool BusinessUnitScoping { get; set; }

    public int Retries { get; set; }
}

public sealed class SegmentDiagnosticIdentifiers
{
    public string? SubscriptionId { get; set; }

    public string? ResourceGroup { get; set; }

    public string? WorkspaceId { get; set; }

    public string? LakehouseId { get; set; }
}

public sealed class SegmentCountStage
{
    public int Order { get; set; }

    public string Label { get; set; } = string.Empty;

    public string Detail { get; set; } = string.Empty;

    public long Count { get; set; }
}

public sealed class SegmentSourceTable
{
    public string Schema { get; set; } = string.Empty;

    public string Table { get; set; } = string.Empty;
}
#endif
