var builder = WebApplication.CreateBuilder(args);
builder.Services.AddSingleton<FabricRuntimeOptions>();
builder.Services.AddSingleton<FabricConnectionFactory>();

// One shared credential so the Fabric REST/ARM access token it caches after
// startup warm-up (see FabricSqlWarmupService) remains usable by the first
// real requests instead of being discarded with a fresh credential per
// transient FabricCapacityService/FabricDataverseShortcutService instance.
builder.Services.AddSingleton<TokenCredential>(_ => new DefaultAzureCredential());
// Short-TTL cache of the last confirmed-Active capacity state, shared across
// every transient FabricCapacityService instance so startup warm-up and the
// two immediate endpoint operations that race it do not each pay for a
// separate ARM call when the capacity was just confirmed active.
builder.Services.AddSingleton<FabricCapacityReadinessCache>();
builder.Services.AddHttpClient<FabricDataverseShortcutService>();
builder.Services.AddHttpClient<FabricCapacityService>();
builder.Services.AddHttpClient();
builder.Services.AddTransient<IFabricCapacityReadiness>(
    sp => sp.GetRequiredService<FabricCapacityService>());
builder.Services.AddTransient<IFabricTokenWarmup>(
    sp => sp.GetRequiredService<FabricDataverseShortcutService>());
builder.Services.AddTransient<FabricDependencyReconciler>();
builder.Services.AddTransient<IFabricDependencyReconciler>(
    sp => sp.GetRequiredService<FabricDependencyReconciler>());
builder.Services.AddTransient<SegmentCountRequestCoordinator>();
builder.Services.AddSingleton<SegmentDependencyValidationQueue>();
builder.Services.AddSingleton<ISegmentDependencyValidationQueue>(
    sp => sp.GetRequiredService<SegmentDependencyValidationQueue>());
builder.Services.AddHostedService(
    sp => sp.GetRequiredService<SegmentDependencyValidationQueue>());
builder.Services.AddSingleton<SegmentPreviewSetupService>();


// The setup page polls /api/health from the Dataverse origin while it waits for
// App Service to mount the deployed package. Only that endpoint is opened, only
// to the configured Dataverse origin, and only for GET.
var setupHealthOrigins = SetupHealthCors.ResolveOrigins(
    builder.Configuration["DATAVERSE_ENVIRONMENT_URL"]);
builder.Services.AddCors(options =>
{
    options.AddPolicy(SetupHealthCors.PolicyName, policy =>
    {
        if (setupHealthOrigins.Length == 0)
        {
            policy.DisallowCredentials();
            return;
        }

        policy.WithOrigins(setupHealthOrigins)
            .WithMethods("GET")
            .WithHeaders("Accept")
            .DisallowCredentials();
    });
    options.AddPolicy(SetupHealthCors.KeyCheckPolicyName, policy =>
    {
        if (setupHealthOrigins.Length == 0)
        {
            policy.DisallowCredentials();
            return;
        }

        policy
            .WithOrigins(setupHealthOrigins)
            .WithMethods("GET")
            .WithHeaders("Accept", "x-api-key")
            .DisallowCredentials();
    });
});

var app = builder.Build();
app.UseCors();
app.MapGet("/api/health", () => Results.Ok(new { status = "ok", utc = DateTime.UtcNow }))
    .RequireCors(SetupHealthCors.PolicyName);
app.MapGet(
        "/api/setup/key-check",
        (HttpRequest request) =>
            HasValidApiKey(request)
                ? Results.Ok(new { status = "ok", apiKeyAccepted = true, utc = DateTime.UtcNow })
                : Unauthorized())
    .RequireCors(SetupHealthCors.KeyCheckPolicyName);
app.MapPost(
    "/api/segment-counts",
    async (
        HttpRequest request,
        SegmentCountRequest input,
        SegmentCountRequestCoordinator coordinator,
        ILogger<Program> logger,
        CancellationToken cancellationToken) =>
    {
        var apiTimer = Stopwatch.StartNew();
        if (!HasValidApiKey(request))
        {
            return Results.Json(
                new ApiError
                {
                    Code = "unauthorized",
                    Message = "The API key is missing or invalid."
                },
                statusCode: StatusCodes.Status401Unauthorized);
        }
        try
        {
            input.Diagnostics = SegmentDiagnosticsFactory.Create(
                input,
                request.HttpContext.TraceIdentifier);
            var response = await coordinator.HandleAsync(input, cancellationToken);
            input.Diagnostics.TimingsMs.ApiTotal = apiTimer.Elapsed.TotalMilliseconds;
            return Results.Ok(response);
        }
        catch (SegmentQueryException exception)
        {
            input.Diagnostics.TimingsMs.ApiTotal = apiTimer.Elapsed.TotalMilliseconds;
            input.Diagnostics.ErrorCode = exception.Code;
            logger.LogWarning(
                exception,
                "Segment-count query rejected with code {Code}.",
                exception.Code);
            return Results.Json(
                new ApiError
                {
                    Code = exception.Code,
                    Message = exception.Message,
                    Detail = exception.Detail,
                    Diagnostics = input.Diagnostics
                },
                statusCode: StatusCodes.Status422UnprocessableEntity);
        }
        catch (FabricQueryException exception)
        {
            input.Diagnostics.TimingsMs.ApiTotal = apiTimer.Elapsed.TotalMilliseconds;
            input.Diagnostics.ErrorCode = exception.Code;
            logger.LogWarning(
                exception,
                "Segment-count interaction rejected with code {Code}.",
                exception.Code);
            return Results.Json(
                new ApiError
                {
                    Code = exception.Code,
                    Message = exception.Message,
                    Detail = exception.Detail,
                    Diagnostics = input.Diagnostics
                },
                statusCode: StatusCodes.Status422UnprocessableEntity);
        }
        catch (Exception exception)
        {
            input.Diagnostics.TimingsMs.ApiTotal = apiTimer.Elapsed.TotalMilliseconds;
            input.Diagnostics.ErrorCode = "execution_failed";
            logger.LogError(exception, "Segment-count execution failed.");
            return Results.Json(
                new ApiError
                {
                    Code = "execution_failed",
                    Message = "The Fabric segment evaluation failed.",
                    Diagnostics = input.Diagnostics
                },
                statusCode: StatusCodes.Status500InternalServerError);
        }
    });

static IResult Unauthorized()
{
    return Results.Json(
        new ApiError
        {
            Code = "unauthorized",
            Message = "The API key is missing or invalid."
        },
        statusCode: StatusCodes.Status401Unauthorized);
}

static bool HasValidApiKey(HttpRequest request)
{
    var expected = Environment.GetEnvironmentVariable("BEHAVIORAL_API_KEY");
    var supplied = request.Headers["x-api-key"].FirstOrDefault();
    if (string.IsNullOrEmpty(expected) || string.IsNullOrEmpty(supplied))
    {
        return false;
    }

    var expectedBytes = Encoding.UTF8.GetBytes(expected);
    var suppliedBytes = Encoding.UTF8.GetBytes(supplied);
    return expectedBytes.Length == suppliedBytes.Length &&
        CryptographicOperations.FixedTimeEquals(expectedBytes, suppliedBytes);
}
