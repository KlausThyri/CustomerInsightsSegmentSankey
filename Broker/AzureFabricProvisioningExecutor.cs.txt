/// <summary>
/// Performs the Azure Resource Manager and Fabric REST work with the delegated
/// permissions of the signed-in administrator. The call shapes mirror the browser
/// engine's direct client exactly, so both paths converge on the same resources.
/// </summary>
public sealed class AzureFabricProvisioningExecutor : IProvisioningExecutor
{
    private const string ArmRoot = "https://management.azure.com";
    private const string ArmApiVersion = "2021-04-01";
    private const string WebApiVersion = "2022-03-01";
    private const string ContainerApiVersion = "2023-05-01";
    private const string FabricRoot = "https://api.fabric.microsoft.com/v1";
    private const string ArmScope = "https://management.azure.com/.default";
    private const string FabricScope = "https://api.fabric.microsoft.com/.default";

    /// <summary>Attempts/delay mirror the browser engine's waitForPackageCopy defaults (7 minutes).</summary>
    private const int PackageCopyAttempts = 210;
    private static readonly TimeSpan PackageCopyPollDelay = TimeSpan.FromSeconds(2);
    private const int PackageCopyLogCharacterLimit = 4000;

    /// <summary>Attempts/delay mirror the browser engine's HEALTH_ATTEMPTS/HEALTH_DELAY_MS defaults (5 minutes).</summary>
    private const int HealthAttempts = 60;
    private static readonly TimeSpan HealthPollDelay = TimeSpan.FromSeconds(5);

    private readonly ITokenClient _tokens;
    private readonly HttpClient _http;
    private readonly IArmTemplateSource _template;
    private readonly TimeProvider _time;

    public AzureFabricProvisioningExecutor(ITokenClient tokens, HttpClient http, IArmTemplateSource template, TimeProvider? time = null)
    {
        _tokens = tokens;
        _http = http;
        _template = template;
        _time = time ?? TimeProvider.System;
    }

    public Task<StepOutcome> ExecuteAsync(string stepId, ProvisioningContext context, CancellationToken cancellationToken) => stepId switch
    {
        "fabric-discovery" => DiscoverFabricAsync(context, cancellationToken),
        "fabric-notebook" => Task.FromResult(NotebookOutcome(context)),
        "azure-infra" => DeployAzureAsync(context, cancellationToken),
        "fabric-permissions" => GrantFabricAccessAsync(context, cancellationToken),
        "fabric-connection-permissions" => GrantConnectionAccessAsync(context, cancellationToken),
        "azure-app" => ConfigureWebAppAsync(context, cancellationToken),
        _ => Task.FromResult(StepOutcome.Skipped("No service-side work is defined for " + stepId + "."))
    };

    // ------------------------------------------------------------------ Fabric

    private async Task<StepOutcome> DiscoverFabricAsync(ProvisioningContext context, CancellationToken cancellationToken)
    {
        var fabric = context.Request.Fabric ?? throw new InvalidOperationException("The run carries no Fabric target.");
        var requestedWorkspaceId = Trim(fabric.WorkspaceId);
        var workspaceName = Trim(fabric.WorkspaceName);
        var workspaces = await FabricCollectionAsync(context, "workspaces", cancellationToken).ConfigureAwait(false);
        var workspace = workspaces
            .FirstOrDefault(item =>
                requestedWorkspaceId is not null &&
                string.Equals(Text(item, "id"), requestedWorkspaceId, StringComparison.OrdinalIgnoreCase) ||
                workspaceName is not null &&
                string.Equals(Text(item, "displayName"), workspaceName, StringComparison.OrdinalIgnoreCase));
        var workspaceId = workspace.ValueKind == JsonValueKind.Undefined ? null : Text(workspace, "id");

        if (workspaceId is null)
        {
            var name = workspaceName ?? throw new InvalidOperationException("Supply a Fabric workspace id or a workspace name.");
            var capacityId = Trim(fabric.CapacityId);
            if (capacityId is null)
            {
                return StepOutcome.ActionRequired(
                    "The Fabric workspace '" + name + "' does not exist and no capacity id was supplied.",
                    "Assign a Fabric capacity and supply its id, or create the workspace '" + name + "' manually, then install again.");
            }

            var created = await FabricAsync(context, HttpMethod.Post, "workspaces",
                new { displayName = name, capacityId }, cancellationToken).ConfigureAwait(false);
            workspaceId = Text(created, "id");
            workspace = created;
        }

        var requestedLakehouseId = Trim(fabric.ServingLakehouseId);
        var lakehouseName = Trim(fabric.ServingLakehouseName) ?? "SegmentServing";
        var lakehouses = await FabricCollectionAsync(context, "workspaces/" + workspaceId + "/lakehouses", cancellationToken).ConfigureAwait(false);
        var lakehouseId = lakehouses
            .FirstOrDefault(item =>
                requestedLakehouseId is not null &&
                string.Equals(Text(item, "id"), requestedLakehouseId, StringComparison.OrdinalIgnoreCase) ||
                string.Equals(Text(item, "displayName"), lakehouseName, StringComparison.OrdinalIgnoreCase))
            is { } foundLakehouse ? Text(foundLakehouse, "id") : null;
        if (lakehouseId is null)
        {
            var created = await FabricAsync(context, HttpMethod.Post, "workspaces/" + workspaceId + "/lakehouses",
                new { displayName = lakehouseName }, cancellationToken).ConfigureAwait(false);
            lakehouseId = Text(created, "id");
        }

        context.Outputs["workspaceId"] = workspaceId;
        context.Outputs["servingLakehouseId"] = lakehouseId;
        var workspaceCapacityId = Text(workspace, "capacityId") ?? Trim(fabric.CapacityId);
        var capacityResourceId = Trim(fabric.CapacityResourceId) ??
            await ResolveCapacityResourceIdAsync(
                context,
                Require(workspaceCapacityId, "capacityId"),
                cancellationToken).ConfigureAwait(false);
        fabric.CapacityResourceId = capacityResourceId;
        context.Outputs["capacityResourceId"] = capacityResourceId;

        var source = await DiscoverDataverseSourceAsync(
            context,
            workspaceId!,
            lakehouses,
            cancellationToken).ConfigureAwait(false);
        if (source is null)
        {
            return StepOutcome.ActionRequired(
                "The Dataverse source Lakehouse could not be discovered.",
                "In Power Apps, verify Link to Microsoft Fabric, then run setup again.");
        }

        fabric.DataverseLakehouseId = source.LakehouseId;
        fabric.DataverseConnectionId = source.ConnectionId;
        fabric.DataverseDeltaFolder = source.DeltaLakeFolder;
        context.Outputs["dataverseLakehouseId"] = source.LakehouseId;

        var sqlEndpoint = await ReadSqlEndpointAsync(context, workspaceId!, lakehouseId!, cancellationToken).ConfigureAwait(false);
        if (sqlEndpoint is not null)
        {
            context.Outputs["fabricSqlServer"] = sqlEndpoint.ConnectionString;
            context.Outputs["fabricSqlDatabase"] = sqlEndpoint.Id;
        }

        return StepOutcome.Succeeded(
            "Fabric workspace, serving lakehouse, and Dataverse source are ready.",
            new Dictionary<string, string?>
            {
                ["workspaceId"] = workspaceId,
                ["servingLakehouseId"] = lakehouseId,
                ["dataverseLakehouseId"] = source.LakehouseId,
                ["fabricSqlServer"] = sqlEndpoint?.ConnectionString,
                ["fabricSqlDatabase"] = sqlEndpoint?.Id
            });
    }

    private async Task<DataverseSource?> DiscoverDataverseSourceAsync(
        ProvisioningContext context,
        string workspaceId,
        IReadOnlyCollection<JsonElement> lakehouses,
        CancellationToken cancellationToken)
    {
        var environmentHost = new Uri(
            context.Request.Dataverse?.EnvironmentUrl ??
            throw new InvalidOperationException("The Dataverse environment URL is missing.")).Host;
        var requestedSourceId = Trim(context.Request.Fabric?.DataverseLakehouseId);
        var candidates = lakehouses
            .OrderBy(lakehouse =>
                requestedSourceId is not null &&
                string.Equals(
                    Text(lakehouse, "id"),
                    requestedSourceId,
                    StringComparison.OrdinalIgnoreCase)
                    ? 0
                    : 1);
        DataverseSource? fallback = null;
        foreach (var lakehouse in candidates)
        {
            var candidateId = Text(lakehouse, "id");
            if (string.IsNullOrWhiteSpace(candidateId))
            {
                continue;
            }

            List<JsonElement> shortcuts;
            try
            {
                shortcuts = await FabricCollectionAsync(
                    context,
                    "workspaces/" + workspaceId + "/items/" + candidateId +
                    "/shortcuts?parentPath=Tables",
                    cancellationToken).ConfigureAwait(false);
            }
            catch (ProvisioningRequestException exception)
                when (exception.StatusCode is HttpStatusCode.BadRequest or
                    HttpStatusCode.NotFound)
            {
                continue;
            }

            foreach (var shortcut in shortcuts)
            {
                if (!IsRootDataverseShortcut(
                        shortcut,
                        environmentHost,
                        out var name,
                        out var connectionId,
                        out var deltaLakeFolder))
                {
                    continue;
                }

                var source = new DataverseSource(
                    candidateId,
                    connectionId,
                    deltaLakeFolder);
                if (string.Equals(name, "contact", StringComparison.OrdinalIgnoreCase))
                {
                    return source;
                }

                fallback ??= source;
            }
        }

        return fallback;
    }

    public static bool IsRootDataverseShortcut(
        JsonElement shortcut,
        string environmentHost,
        out string name,
        out string connectionId,
        out string deltaLakeFolder)
    {
        name = "";
        connectionId = "";
        deltaLakeFolder = "";
        if (!shortcut.TryGetProperty("path", out var path) ||
            !string.Equals(
                path.GetString()?.Trim('/'),
                "Tables",
                StringComparison.OrdinalIgnoreCase) ||
            !shortcut.TryGetProperty("name", out var nameElement) ||
            !shortcut.TryGetProperty("target", out var target) ||
            !target.TryGetProperty("dataverse", out var dataverse))
        {
            return false;
        }

        name = nameElement.GetString() ?? "";
        connectionId = Text(dataverse, "connectionId") ?? "";
        deltaLakeFolder = Text(dataverse, "deltaLakeFolder") ?? "";
        var environmentDomain = Text(dataverse, "environmentDomain");
        return Uri.TryCreate(environmentDomain, UriKind.Absolute, out var environmentUri) &&
            string.Equals(
                environmentUri.Host,
                environmentHost,
                StringComparison.OrdinalIgnoreCase) &&
            !string.IsNullOrWhiteSpace(name) &&
            !string.IsNullOrWhiteSpace(connectionId) &&
            !string.IsNullOrWhiteSpace(deltaLakeFolder);
    }

    private async Task<SqlEndpoint?> ReadSqlEndpointAsync(ProvisioningContext context, string workspaceId, string lakehouseId, CancellationToken cancellationToken)
    {
        try
        {
            var lakehouse = await FabricAsync(context, HttpMethod.Get, "workspaces/" + workspaceId + "/lakehouses/" + lakehouseId, null, cancellationToken).ConfigureAwait(false);
            if (lakehouse.TryGetProperty("properties", out var properties) &&
                properties.TryGetProperty("sqlEndpointProperties", out var endpoint) &&
                endpoint.TryGetProperty("connectionString", out var connection) &&
                endpoint.TryGetProperty("id", out var id))
            {
                return new SqlEndpoint(
                    id.GetString() ?? "",
                    connection.GetString() ?? "");
            }
        }
        catch (InvalidOperationException)
        {
            // The endpoint is provisioned asynchronously by Fabric; absence is not an error.
        }
        return null;
    }

    private static StepOutcome NotebookOutcome(ProvisioningContext context)
    {
        if (context.Request.Options?.SkipNotebook == true)
        {
            return StepOutcome.Skipped("The bootstrap notebook was skipped on request.");
        }

        return StepOutcome.ActionRequired(
            "The serving bootstrap notebook was not published by the service.",
            "Import Fabric/segment_serving_bootstrap notebook into the workspace and schedule it daily. " +
            "The service does not ship the notebook definition, so this step stays with the workspace administrator.");
    }

    private async Task<StepOutcome> GrantFabricAccessAsync(ProvisioningContext context, CancellationToken cancellationToken)
    {
        var workspaceId = context.Outputs.TryGetValue("workspaceId", out var value) ? value : null;
        var principalId = context.Outputs.TryGetValue("principalId", out var principal) ? principal : null;

        if (workspaceId is null || principalId is null)
        {
            return StepOutcome.ActionRequired(
                "The managed identity could not be granted access because the workspace or identity is unknown.",
                "Add the Web App managed identity as a Contributor of the Fabric workspace.");
        }

        try
        {
            var upgraded = false;
            var path = "workspaces/" + Uri.EscapeDataString(workspaceId) + "/roleAssignments";
            var assignments = await FabricCollectionAsync(
                context,
                path,
                cancellationToken).ConfigureAwait(false);
            var exists = assignments.Any(assignment =>
                string.Equals(Text(assignment, "id"), principalId, StringComparison.OrdinalIgnoreCase) ||
                assignment.TryGetProperty("principal", out var assignedPrincipal) &&
                string.Equals(Text(assignedPrincipal, "id"), principalId, StringComparison.OrdinalIgnoreCase));
            if (!exists)
            {
                try
                {
                    await FabricAsync(
                        context,
                        HttpMethod.Post,
                        path,
                        new
                        {
                            principal = new { id = principalId, type = "ServicePrincipal" },
                            role = "Contributor"
                        },
                        cancellationToken).ConfigureAwait(false);
                }
                catch (ProvisioningRequestException exception)
                    when (exception.ResponseBody.Contains(
                        "principal already has a role assigned in the workspace",
                        StringComparison.OrdinalIgnoreCase))
                {
                    exists = true;
                }
            }
            else
            {
                var existing = assignments.First(assignment =>
                    string.Equals(Text(assignment, "id"), principalId, StringComparison.OrdinalIgnoreCase) ||
                    assignment.TryGetProperty("principal", out var assignedPrincipal) &&
                    string.Equals(Text(assignedPrincipal, "id"), principalId, StringComparison.OrdinalIgnoreCase));
                var role = Text(existing, "role");
                if (role is not ("Admin" or "Member" or "Contributor"))
                {
                    var assignmentId = Text(existing, "id") ?? principalId;
                    await FabricAsync(
                        context,
                        HttpMethod.Patch,
                        path + "/" + Uri.EscapeDataString(assignmentId),
                        new { role = "Contributor" },
                        cancellationToken).ConfigureAwait(false);
                    upgraded = true;
                }
            }
            var capacityResourceId = Trim(context.Request.Fabric?.CapacityResourceId);
            if (capacityResourceId is null)
            {
                return StepOutcome.ActionRequired(
                    "The Azure Fabric capacity resource is unknown.",
                    "Run Fabric discovery again, then install again.");
            }
            await EnsureCapacityRoleAssignmentAsync(
                context,
                capacityResourceId,
                principalId,
                cancellationToken).ConfigureAwait(false);
            return StepOutcome.Succeeded(
                upgraded
                    ? "The workspace role was upgraded and capacity start permission was assigned."
                    : exists
                    ? "The managed identity already has workspace access and capacity start permission."
                    : "The managed identity is a workspace Contributor and can start the capacity.");
        }
        catch (InvalidOperationException exception)
        {
            return StepOutcome.ActionRequired(
                "The workspace role assignment was refused: " + exception.Message,
                "Enable the Fabric tenant setting 'Service principals can use Fabric APIs' for a security group that contains the Web App managed identity, then install again.");
        }
    }

    private async Task<StepOutcome> GrantConnectionAccessAsync(
        ProvisioningContext context,
        CancellationToken cancellationToken)
    {
        var connectionId = Trim(context.Request.Fabric?.DataverseConnectionId);
        var principalId = context.Outputs.TryGetValue("principalId", out var principal) ? principal : null;
        if (connectionId is null || principalId is null)
        {
            return StepOutcome.ActionRequired(
                "The managed identity could not be granted connection access because the connection or identity is unknown.",
                "Select a Fabric Dataverse connection and run the installation again.");
        }

        try
        {
            var assignments = await FabricCollectionAsync(
                context,
                "connections/" + Uri.EscapeDataString(connectionId) + "/roleAssignments",
                cancellationToken).ConfigureAwait(false);
            var exists = assignments.Any(assignment =>
                string.Equals(Text(assignment, "id"), principalId, StringComparison.OrdinalIgnoreCase) ||
                assignment.TryGetProperty("principal", out var assignedPrincipal) &&
                string.Equals(Text(assignedPrincipal, "id"), principalId, StringComparison.OrdinalIgnoreCase));
            if (!exists)
            {
                var path = "connections/" + Uri.EscapeDataString(connectionId) + "/roleAssignments";
                try
                {
                    await FabricAsync(
                        context,
                        HttpMethod.Post,
                        path,
                        new
                        {
                            principal = new { id = principalId, type = "ServicePrincipal" },
                            role = "User"
                        },
                        cancellationToken).ConfigureAwait(false);
                }
                catch (ProvisioningRequestException postException)
                {
                    try
                    {
                        await FabricAsync(
                            context,
                            HttpMethod.Get,
                            path + "/" + Uri.EscapeDataString(principalId),
                            null,
                            cancellationToken).ConfigureAwait(false);
                        exists = true;
                    }
                    catch (ProvisioningRequestException)
                    {
                        throw postException;
                    }
                }
            }

            return StepOutcome.Succeeded(
                exists
                    ? "The managed identity already has access to the Dataverse connection."
                    : "The managed identity received User access to the Dataverse connection.");
        }
        catch (InvalidOperationException exception)
        {
            return StepOutcome.ActionRequired(
                "The Dataverse connection role assignment was refused: " + exception.Message,
                "Workspace Admin, Fabric Admin, and Power Platform Admin are not sufficient. In Fabric, open Settings > Manage connections and gateways, select the Dataverse connection, and have its connection owner grant the signed-in user Owner or UserWithReshare access.");
        }
    }

    private async Task<StepOutcome> DeployAzureAsync(ProvisioningContext context, CancellationToken cancellationToken)
    {
        var azure = context.Request.Azure ?? throw new InvalidOperationException("The run carries no Azure target.");
        var subscriptionId = Require(azure.SubscriptionId, "subscriptionId");
        var resourceGroup = Require(azure.ResourceGroup, "resourceGroup");
        var location = Trim(azure.Location) ?? "westeurope";
        var webAppName = Require(azure.WebAppName, "webAppName");

        var resourceGroupPath =
            "/subscriptions/" + subscriptionId + "/resourcegroups/" + Uri.EscapeDataString(resourceGroup);
        try
        {
            await ArmAsync(
                context,
                HttpMethod.Get,
                resourceGroupPath,
                null,
                ArmApiVersion,
                cancellationToken).ConfigureAwait(false);
        }
        catch (ProvisioningRequestException exception) when (exception.StatusCode == System.Net.HttpStatusCode.NotFound)
        {
            await ArmAsync(
                context,
                HttpMethod.Put,
                resourceGroupPath,
                new { location },
                ArmApiVersion,
                cancellationToken).ConfigureAwait(false);
        }

        var template = _template.Read();
        if (template is null)
        {
            return StepOutcome.ActionRequired(
                "The service has no ARM template available.",
                "Deploy deployment/azure/main.bicep into the target resource group, then install again.");
        }

        var fabric = context.Request.Fabric;
        var package = context.Request.ApiPackage;
        if (package is null || !package.IsUsable)
        {
            // The template writes the application settings as a complete set, so a
            // package-less deployment would remove WEBSITE_RUN_FROM_PACKAGE from a
            // Web App that already serves the API. Nothing is deployed instead.
            return StepOutcome.ActionRequired(
                "No verified API package was supplied, so the Azure deployment was not started.",
                "The setup page must send an https package URL together with its SHA-256 digest. " +
                "Set klth_SetupApiPackageUrl to '<https url> <64 hex sha-256>' on the setup page and install again.");
        }
        if (Trim(fabric?.DataverseLakehouseId) is null)
        {
            return StepOutcome.ActionRequired(
                "The Dataverse source Lakehouse ID is missing, so the Azure deployment was not started.",
                "Run Fabric discovery again and select the Lakehouse created by Link to Microsoft Fabric.");
        }

        var blobName = package.ResolvedBlobName;
        var parameters = new Dictionary<string, object?>
        {
            ["location"] = location,
            ["webAppName"] = webAppName,
            ["fabricSqlServer"] = context.Outputs.TryGetValue("fabricSqlServer", out var sql) ? sql ?? string.Empty : string.Empty,
            ["fabricSqlDatabase"] = context.Outputs.TryGetValue("fabricSqlDatabase", out var database) ? database ?? string.Empty : string.Empty,
            ["fabricWorkspaceId"] = context.Outputs.TryGetValue("workspaceId", out var workspace) ? workspace ?? string.Empty : string.Empty,
            ["fabricServingLakehouseId"] = context.Outputs.TryGetValue("servingLakehouseId", out var lakehouse) ? lakehouse ?? string.Empty : string.Empty,
            ["fabricCapacityResourceId"] = Trim(fabric?.CapacityResourceId) ?? string.Empty,
            ["fabricDataverseLakehouseId"] = Trim(fabric?.DataverseLakehouseId) ?? string.Empty,
            ["fabricDataverseConnectionId"] = Trim(fabric?.DataverseConnectionId) ?? string.Empty,
            ["fabricDataverseDeltaFolder"] = Trim(fabric?.DataverseDeltaFolder) ?? "deltalake",
            ["dataverseEnvironmentUrl"] = Trim(context.Request.Dataverse?.EnvironmentUrl) ?? string.Empty,
            ["behavioralApiKey"] = context.Request.Secrets?.BehavioralApiKey ?? string.Empty,
            ["requiredDataverseTables"] = string.Join(',', context.Request.Dataverse?.RequiredTables ?? Array.Empty<string>()),
            ["apiPackageUrl"] = package.Url!.Trim(),
            ["apiPackageSha256"] = package.Sha256!.Trim().ToLowerInvariant(),
            ["apiPackageBlobName"] = blobName,
            ["apiPackageVersion"] = Trim(package.Version) ?? string.Empty
        };

        context.Outputs["packageBlob"] = blobName;
        context.Outputs["packageSha256"] = package.Sha256!.Trim().ToLowerInvariant();

        var wrapped = parameters.ToDictionary(pair => pair.Key, pair => (object)new { value = pair.Value });
        var path = "/subscriptions/" + subscriptionId + "/resourcegroups/" + Uri.EscapeDataString(resourceGroup) +
                   "/providers/Microsoft.Resources/deployments/segment-preview";

        await ArmAsync(context, HttpMethod.Put, path,
            new { properties = new { mode = "Incremental", template, parameters = wrapped } },
            ArmApiVersion, cancellationToken).ConfigureAwait(false);

        for (var poll = 0; poll < 240; poll++)
        {
            var snapshot = await ArmAsync(context, HttpMethod.Get, path, null, ArmApiVersion, cancellationToken).ConfigureAwait(false);
            var state = snapshot.TryGetProperty("properties", out var properties) && properties.TryGetProperty("provisioningState", out var provisioning)
                ? provisioning.GetString()
                : null;

            if (state == "Succeeded")
            {
                if (properties.TryGetProperty("outputs", out var outputs))
                {
                    foreach (var output in outputs.EnumerateObject())
                    {
                        if (output.Value.TryGetProperty("value", out var outputValue) && outputValue.ValueKind == JsonValueKind.String)
                        {
                            context.Outputs[output.Name] = outputValue.GetString();
                        }
                    }
                }

                // The template deploys a container group that downloads and verifies the
                // package into the customer's own storage account. A Web App can only be
                // trusted to run that package once Azure itself reports the copy finished
                // cleanly, so this is checked before the deployment is ever reported done.
                await VerifyPackageCopyAsync(context, subscriptionId, resourceGroup, webAppName, cancellationToken).ConfigureAwait(false);

                ApplyDeploymentOutputs(context.Outputs, webAppName);
                return StepOutcome.Succeeded("The Azure infrastructure was deployed.",
                    new Dictionary<string, string?>
                    {
                        ["apiBaseUrl"] = context.Outputs["apiBaseUrl"],
                        ["principalId"] = context.Outputs.TryGetValue("principalId", out var mapped) ? mapped : null
                    });
            }

            if (state is "Failed" or "Canceled")
            {
                throw new InvalidOperationException("The Azure deployment finished with state '" + state + "'.");
            }

            await Task.Delay(TimeSpan.FromSeconds(5), _time, cancellationToken).ConfigureAwait(false);
        }

        throw new InvalidOperationException("The Azure deployment did not finish within the expected time.");
    }

    /// <summary>
    /// Waits for the '&lt;webAppName&gt;-package-copy' container instance to reach a
    /// terminal state, retries once by restarting the container group, and fails
    /// with a bounded, sanitized tail of its logs if the retry does not exit 0.
    /// </summary>
    private async Task VerifyPackageCopyAsync(
        ProvisioningContext context,
        string subscriptionId,
        string resourceGroup,
        string webAppName,
        CancellationToken cancellationToken)
    {
        var containerGroupName = webAppName + "-package-copy";
        var path = ContainerGroupPath(subscriptionId, resourceGroup, containerGroupName);

        var copy = await WaitForPackageCopyAsync(context, path, containerGroupName, null, cancellationToken).ConfigureAwait(false);
        if (copy.ExitCode != 0)
        {
            if (!DateTimeOffset.TryParse(
                    copy.StartTime,
                    CultureInfo.InvariantCulture,
                    DateTimeStyles.AssumeUniversal,
                    out _))
            {
                throw new InvalidOperationException(
                    "Azure did not return a valid start time for the failed package-copy container, so Setup cannot verify a retry safely.");
            }

            await RestartContainerGroupAsync(context, path, cancellationToken).ConfigureAwait(false);
            copy = await WaitForPackageCopyAsync(context, path, containerGroupName, copy.StartTime, cancellationToken).ConfigureAwait(false);
        }

        if (copy.ExitCode != 0)
        {
            string logs;
            try
            {
                logs = await GetPackageCopyLogsAsync(context, path, cancellationToken).ConfigureAwait(false);
            }
            catch (ProvisioningRequestException exception)
            {
                logs = "Container logs could not be read: " + exception.Message;
            }

            throw new InvalidOperationException(
                "The Azure package-copy container failed with exit code " +
                (copy.ExitCode?.ToString(CultureInfo.InvariantCulture) ?? "unknown") +
                (copy.DetailStatus.Length > 0 ? " (" + copy.DetailStatus + ")" : string.Empty) +
                ". The Web App was not marked ready because its API package blob may be missing. " +
                "Inspect the '" + containerGroupName + "' container group in resource group " + resourceGroup + "." +
                (logs.Length > 0 ? Environment.NewLine + logs : string.Empty));
        }
    }

    private static string ContainerGroupPath(string subscriptionId, string resourceGroup, string containerGroupName) =>
        "/subscriptions/" + subscriptionId + "/resourceGroups/" + Uri.EscapeDataString(resourceGroup) +
        "/providers/Microsoft.ContainerInstance/containerGroups/" + Uri.EscapeDataString(containerGroupName);

    private async Task<PackageCopyState> GetPackageCopyStateAsync(ProvisioningContext context, string path, CancellationToken cancellationToken)
    {
        var response = await ArmAsync(context, HttpMethod.Get, path, null, ContainerApiVersion, cancellationToken).ConfigureAwait(false);
        var currentState = default(JsonElement);
        if (response.ValueKind == JsonValueKind.Object &&
            response.TryGetProperty("properties", out var properties) &&
            properties.TryGetProperty("containers", out var containers) &&
            containers.ValueKind == JsonValueKind.Array)
        {
            foreach (var container in containers.EnumerateArray())
            {
                if (string.Equals(Text(container, "name"), "copy", StringComparison.OrdinalIgnoreCase) &&
                    container.TryGetProperty("properties", out var containerProperties) &&
                    containerProperties.TryGetProperty("instanceView", out var instanceView) &&
                    instanceView.ValueKind == JsonValueKind.Object &&
                    instanceView.TryGetProperty("currentState", out var state) &&
                    state.ValueKind == JsonValueKind.Object)
                {
                    currentState = state;
                    break;
                }
            }
        }

        if (currentState.ValueKind != JsonValueKind.Object)
        {
            return new PackageCopyState(string.Empty, null, string.Empty, string.Empty);
        }

        var exitCode = currentState.TryGetProperty("exitCode", out var exitCodeElement) && exitCodeElement.ValueKind == JsonValueKind.Number
            ? exitCodeElement.GetInt32()
            : (int?)null;

        return new PackageCopyState(
            Text(currentState, "state") ?? string.Empty,
            exitCode,
            Text(currentState, "detailStatus") ?? string.Empty,
            Text(currentState, "startTime") ?? string.Empty);
    }

    /// <summary>
    /// Polls until the copy container reports a terminated state. When
    /// <paramref name="startedAfter"/> is supplied (the Azure-reported startTime
    /// from the run being retried) a terminated state is only accepted once the
    /// container reports a newer Azure-observed startTime, so a stale terminal
    /// state left over from the failed run can never be mistaken for the retry -
    /// without ever comparing against the broker's own clock.
    /// </summary>
    private async Task<PackageCopyState> WaitForPackageCopyAsync(
        ProvisioningContext context,
        string path,
        string containerGroupName,
        string? startedAfter,
        CancellationToken cancellationToken)
    {
        for (var attempt = 1; attempt <= PackageCopyAttempts; attempt++)
        {
            try
            {
                var latest = await GetPackageCopyStateAsync(context, path, cancellationToken).ConfigureAwait(false);
                var isTerminated = string.Equals(latest.State, "Terminated", StringComparison.OrdinalIgnoreCase);
                var isCurrentRun = startedAfter is null || IsNewerStartTime(latest.StartTime, startedAfter);
                if (isTerminated && isCurrentRun)
                {
                    return latest;
                }
            }
            catch (ProvisioningRequestException exception) when (
                exception.StatusCode == HttpStatusCode.TooManyRequests ||
                (int)exception.StatusCode >= 500)
            {
            }
            catch (HttpRequestException)
            {
            }

            if (attempt < PackageCopyAttempts)
            {
                await Task.Delay(PackageCopyPollDelay, _time, cancellationToken).ConfigureAwait(false);
            }
        }

        throw new InvalidOperationException(
            "Azure did not report a terminal state for package-copy container '" + containerGroupName +
            "' within " + (int)Math.Round(PackageCopyAttempts * PackageCopyPollDelay.TotalSeconds) + " seconds.");
    }

    private static bool IsNewerStartTime(string candidate, string previous) =>
        DateTimeOffset.TryParse(candidate, CultureInfo.InvariantCulture, DateTimeStyles.AssumeUniversal, out var candidateTime) &&
        DateTimeOffset.TryParse(previous, CultureInfo.InvariantCulture, DateTimeStyles.AssumeUniversal, out var previousTime) &&
        candidateTime > previousTime;

    private async Task RestartContainerGroupAsync(ProvisioningContext context, string path, CancellationToken cancellationToken) =>
        await ArmAsync(context, HttpMethod.Post, path + "/restart", null, ContainerApiVersion, cancellationToken).ConfigureAwait(false);

    private async Task<string> GetPackageCopyLogsAsync(ProvisioningContext context, string path, CancellationToken cancellationToken)
    {
        var response = await ArmAsync(context, HttpMethod.Get, path + "/containers/copy/logs?tail=100", null, ContainerApiVersion, cancellationToken)
            .ConfigureAwait(false);
        var content = response.ValueKind == JsonValueKind.Object ? Text(response, "content") : null;
        return SanitizeContainerLogs(content ?? string.Empty);
    }

    /// <summary>Bounds the log tail and strips any SAS-bearing URL query string before it can reach an error message.</summary>
    private static string SanitizeContainerLogs(string logs)
    {
        if (string.IsNullOrEmpty(logs)) return string.Empty;
        var sanitized = Regex.Replace(
            logs,
            @"(https?://[^\s""']+?)\?[^\s""')]*",
            "$1",
            RegexOptions.None,
            TimeSpan.FromSeconds(1));
        return sanitized.Length > PackageCopyLogCharacterLimit ? sanitized[^PackageCopyLogCharacterLimit..] : sanitized;
    }

    private async Task<StepOutcome> ConfigureWebAppAsync(ProvisioningContext context, CancellationToken cancellationToken)
    {
        var azure = context.Request.Azure ?? throw new InvalidOperationException("The run carries no Azure target.");
        var subscriptionId = Require(azure.SubscriptionId, "subscriptionId");
        var resourceGroup = Require(azure.ResourceGroup, "resourceGroup");
        var webAppName = Require(azure.WebAppName, "webAppName");
        var site = "/subscriptions/" + subscriptionId + "/resourceGroups/" + Uri.EscapeDataString(resourceGroup) +
                   "/providers/Microsoft.Web/sites/" + Uri.EscapeDataString(webAppName);

        var package = context.Request.ApiPackage;
        if (package is null || !package.IsUsable)
        {
            return StepOutcome.ActionRequired(
                "No verified API package was supplied, so the Web App cannot be confirmed to run the API.",
                "Send an https package URL together with its SHA-256 digest and install again.");
        }

        var identity = await ArmAsync(context, HttpMethod.Get, site, null, WebApiVersion, cancellationToken).ConfigureAwait(false);
        if (identity.TryGetProperty("identity", out var identityBlock) &&
            identityBlock.TryGetProperty("principalId", out var principalId) &&
            principalId.ValueKind == JsonValueKind.String)
        {
            context.Outputs["principalId"] = principalId.GetString();
        }

        // The deployment, not this service, copied and verified the package. This
        // step reads back what Azure actually applied so a codeless Web App can
        // never be reported as a success.
        var settings = await ArmAsync(context, HttpMethod.Post, site + "/config/appsettings/list", null, WebApiVersion, cancellationToken)
            .ConfigureAwait(false);
        var applied = ReadAppSettings(settings);
        VerifyPackageSettings(applied, package);
        VerifyApiKeySetting(applied, context.Request.Secrets?.BehavioralApiKey);

        await ArmAsync(context, HttpMethod.Post, site + "/restart", null, WebApiVersion, cancellationToken).ConfigureAwait(false);

        var blobUrl = applied.TryGetValue(RunFromPackageSetting, out var deployed) ? deployed : null;
        context.Outputs["packageBlobUrl"] = blobUrl;

        // Settings alone do not prove the API runs: the blob mount and the role
        // assignment that lets the site read it both propagate asynchronously, so
        // this waits for the API's own public health endpoint, then for the same
        // process to accept the key Azure just applied, before the step succeeds.
        var apiBaseUrl = context.Outputs.TryGetValue("apiBaseUrl", out var reportedBaseUrl) && !string.IsNullOrWhiteSpace(reportedBaseUrl)
            ? reportedBaseUrl!
            : NormalizeApiBaseUrl(webAppName + ".azurewebsites.net");

        var health = await PollApiHealthAsync(apiBaseUrl, cancellationToken).ConfigureAwait(false);
        if (!health.Ok)
        {
            throw new InvalidOperationException(
                "The API package was deployed but " + apiBaseUrl.TrimEnd('/') + "/health did not answer within " +
                FormatMinutes(HealthAttempts, HealthPollDelay) + " minutes (" + (health.Error ?? "no response") + "). " +
                "App Service may still be mounting the package or the storage role assignment may not have propagated. " +
                "Wait a few minutes and install again; the step is safe to repeat.");
        }

        // The API key is never written to a log or an exception message here.
        var apiKey = context.Request.Secrets?.BehavioralApiKey ?? string.Empty;
        var keyCheck = await PollApiKeyCheckAsync(apiBaseUrl, apiKey, cancellationToken).ConfigureAwait(false);
        if (!keyCheck.Ok)
        {
            throw new InvalidOperationException(
                "The API package is running, but the active Web App process did not accept the API key that Azure applied within " +
                FormatMinutes(HealthAttempts, HealthPollDelay) + " minutes (" + (keyCheck.Error ?? "no response") + "). " +
                "App Service may still be recycling an older worker. Wait a few minutes and install again; the provisioning " +
                "service will reuse the existing key instead of rotating it.");
        }

        return StepOutcome.Succeeded(
            "The Web App runs API package " + (Trim(package.Version) ?? package.ResolvedBlobName) +
            " from the customer's own storage account, was restarted, answered /health after " + health.Attempts +
            (health.Attempts == 1 ? " attempt, and" : " attempts, and") + " accepted its API key after " + keyCheck.Attempts +
            (keyCheck.Attempts == 1 ? " attempt." : " attempts."),
            new Dictionary<string, string?>
            {
                ["principalId"] = context.Outputs.TryGetValue("principalId", out var value) ? value : null,
                ["packageBlobUrl"] = blobUrl
            });
    }

    private static string FormatMinutes(int attempts, TimeSpan delay) =>
        Math.Round(attempts * delay.TotalSeconds / 60, MidpointRounding.AwayFromZero).ToString(CultureInfo.InvariantCulture);

    private sealed record ApiProbeResult(bool Ok, int Attempts, string? Error);

    /// <summary>
    /// Polls the deployed API until it answers its own health endpoint. A Web App
    /// that has just been restarted needs time to mount the package blob, so a
    /// first failure means "not ready yet" rather than "broken".
    /// </summary>
    private async Task<ApiProbeResult> PollApiHealthAsync(string apiBaseUrl, CancellationToken cancellationToken)
    {
        var url = apiBaseUrl.TrimEnd('/') + "/health";
        string? lastError = null;
        for (var attempt = 1; attempt <= HealthAttempts; attempt++)
        {
            if (attempt > 1)
            {
                await Task.Delay(HealthPollDelay, _time, cancellationToken).ConfigureAwait(false);
            }

            try
            {
                using var request = new HttpRequestMessage(HttpMethod.Get, url);
                request.Headers.Accept.Add(new MediaTypeWithQualityHeaderValue("application/json"));
                using var response = await _http.SendAsync(request, cancellationToken).ConfigureAwait(false);
                if (response.IsSuccessStatusCode)
                {
                    return new ApiProbeResult(true, attempt, null);
                }

                lastError = "HTTP " + (int)response.StatusCode;
            }
            catch (HttpRequestException exception)
            {
                lastError = exception.Message;
            }
        }

        return new ApiProbeResult(false, HealthAttempts, lastError);
    }

    /// <summary>
    /// Waits until the active Web App process accepts the key Azure just applied.
    /// The unauthenticated health endpoint can still be served by an older warm
    /// worker while App Service is recycling after a settings change, so this
    /// checks /setup/key-check separately with the same API key header the
    /// Dataverse plugin sends, and never logs the key itself.
    /// </summary>
    private async Task<ApiProbeResult> PollApiKeyCheckAsync(string apiBaseUrl, string apiKey, CancellationToken cancellationToken)
    {
        var url = apiBaseUrl.TrimEnd('/') + "/setup/key-check";
        string? lastError = null;
        for (var attempt = 1; attempt <= HealthAttempts; attempt++)
        {
            if (attempt > 1)
            {
                await Task.Delay(HealthPollDelay, _time, cancellationToken).ConfigureAwait(false);
            }

            try
            {
                using var request = new HttpRequestMessage(HttpMethod.Get, url);
                request.Headers.Accept.Add(new MediaTypeWithQualityHeaderValue("application/json"));
                request.Headers.TryAddWithoutValidation("x-api-key", apiKey);
                using var response = await _http.SendAsync(request, cancellationToken).ConfigureAwait(false);
                if (response.IsSuccessStatusCode)
                {
                    var text = await response.Content.ReadAsStringAsync(cancellationToken).ConfigureAwait(false);
                    if (!string.IsNullOrWhiteSpace(text))
                    {
                        using var document = JsonDocument.Parse(text);
                        if (document.RootElement.ValueKind == JsonValueKind.Object &&
                            document.RootElement.TryGetProperty("apiKeyAccepted", out var accepted) &&
                            accepted.ValueKind == JsonValueKind.True)
                        {
                            return new ApiProbeResult(true, attempt, null);
                        }
                    }
                }

                lastError = "HTTP " + (int)response.StatusCode;
            }
            catch (HttpRequestException exception)
            {
                lastError = exception.Message;
            }
        }

        return new ApiProbeResult(false, HealthAttempts, lastError);
    }

    internal const string RunFromPackageSetting = "WEBSITE_RUN_FROM_PACKAGE";
    internal const string PackageIdentitySetting = "WEBSITE_RUN_FROM_PACKAGE_BLOB_MI_RESOURCE_ID";
    internal const string PackageShaSetting = "SEGMENT_PREVIEW_PACKAGE_SHA256";
    internal const string PackageContainer = "segment-preview-api";
    internal const string ApiKeySetting = "BEHAVIORAL_API_KEY";

    /// <summary>Flattens the ARM app-settings response into a plain dictionary.</summary>
    public static Dictionary<string, string?> ReadAppSettings(JsonElement response)
    {
        var settings = new Dictionary<string, string?>(StringComparer.Ordinal);
        if (response.ValueKind == JsonValueKind.Object &&
            response.TryGetProperty("properties", out var properties) &&
            properties.ValueKind == JsonValueKind.Object)
        {
            foreach (var setting in properties.EnumerateObject())
            {
                settings[setting.Name] = setting.Value.ValueKind == JsonValueKind.String ? setting.Value.GetString() : null;
            }
        }

        return settings;
    }

    /// <summary>
    /// Fails the run unless the Web App really reads the pinned package from
    /// customer-owned storage with its own identity and no expiring signature.
    /// </summary>
    public static void VerifyPackageSettings(IReadOnlyDictionary<string, string?> settings, ApiPackage package)
    {
        var blobName = package.ResolvedBlobName;
        var deployed = settings.TryGetValue(RunFromPackageSetting, out var value) ? value ?? string.Empty : string.Empty;

        if (deployed.IndexOf("/" + PackageContainer + "/" + blobName, StringComparison.Ordinal) < 0)
        {
            throw new InvalidOperationException(
                "The Web App does not run the expected API package. Its " + RunFromPackageSetting + " setting is " +
                (deployed.Length == 0 ? "empty" : deployed) + " instead of a blob named " + blobName + ".");
        }

        if (deployed.Contains('?', StringComparison.Ordinal))
        {
            throw new InvalidOperationException(
                "The Web App reads the API package through a shared access signature, which expires.");
        }

        var digest = (package.Sha256 ?? string.Empty).Trim().ToLowerInvariant();
        var reported = (settings.TryGetValue(PackageShaSetting, out var sha) ? sha ?? string.Empty : string.Empty)
            .Trim().ToLowerInvariant();
        if (!string.Equals(reported, digest, StringComparison.Ordinal))
        {
            throw new InvalidOperationException(
                "The Web App reports API package digest " + (reported.Length == 0 ? "none" : reported) +
                " instead of the pinned " + digest + ".");
        }

        var identity = settings.TryGetValue(PackageIdentitySetting, out var mi) ? mi : null;
        if (!string.Equals(identity, "SystemAssigned", StringComparison.Ordinal))
        {
            throw new InvalidOperationException(
                "The Web App is not configured to read the package with its own managed identity (" +
                PackageIdentitySetting + " is " + (string.IsNullOrEmpty(identity) ? "empty" : identity) + ").");
        }
    }

    public static void VerifyApiKeySetting(
        IReadOnlyDictionary<string, string?> settings,
        string? expectedApiKey)
    {
        var expected = expectedApiKey ?? string.Empty;
        var applied = settings.TryGetValue(ApiKeySetting, out var value)
            ? value ?? string.Empty
            : string.Empty;
        if (!string.Equals(applied, expected, StringComparison.Ordinal))
        {
            throw new InvalidOperationException(
                "Azure did not return the API key that the provisioning service deployed. " +
                "Dataverse was not updated with a key the API would reject.");
        }
    }

    /// <summary>
    /// Returns the API root the Dataverse plugin talks to. The Web App serves
    /// the API under /api/, so a bare host name is not a usable base URL.
    /// </summary>
    public static string NormalizeApiBaseUrl(string? value)
    {
        var text = (value ?? string.Empty).Trim();
        if (text.Length == 0)
        {
            throw new InvalidOperationException("The Azure deployment reported no Web App URL.");
        }
        if (!text.StartsWith("http://", StringComparison.OrdinalIgnoreCase) &&
            !text.StartsWith("https://", StringComparison.OrdinalIgnoreCase))
        {
            text = "https://" + text;
        }
        var uri = new Uri(text, UriKind.Absolute);
        return "https://" + uri.Authority + "/api/";
    }

    /// <summary>
    /// Turns the raw template outputs into the facts the later steps read. The
    /// Fabric role assignment runs before the Web App step, so the managed
    /// identity has to be published under 'principalId' as soon as the
    /// deployment reports it rather than when the Web App is configured.
    /// </summary>
    public static void ApplyDeploymentOutputs(IDictionary<string, string?> outputs, string webAppName)
    {
        ArgumentNullException.ThrowIfNull(outputs);

        if (outputs.TryGetValue("managedIdentityPrincipalId", out var identityPrincipal) &&
            !string.IsNullOrWhiteSpace(identityPrincipal))
        {
            outputs["principalId"] = identityPrincipal;
        }

        var reported = outputs.TryGetValue("webAppUrl", out var webAppUrl) && !string.IsNullOrWhiteSpace(webAppUrl)
            ? webAppUrl
            : "https://" + webAppName + ".azurewebsites.net";
        outputs["apiBaseUrl"] = NormalizeApiBaseUrl(reported);
    }

    private async Task<string> ResolveCapacityResourceIdAsync(
        ProvisioningContext context,
        string capacityId,
        CancellationToken cancellationToken)
    {
        var capacities = await FabricCollectionAsync(
            context,
            "capacities",
            cancellationToken).ConfigureAwait(false);
        var capacityName = capacities
            .Where(item => string.Equals(
                Text(item, "id"),
                capacityId,
                StringComparison.OrdinalIgnoreCase))
            .Select(item => Text(item, "displayName"))
            .FirstOrDefault();
        if (capacityName is null)
        {
            throw new InvalidOperationException(
                "The Fabric capacity could not be read. Verify that the signed-in administrator is a capacity administrator.");
        }

        var preferredSubscription = Require(
            context.Request.Azure?.SubscriptionId,
            "subscriptionId");
        var subscriptionsPayload = await ArmAsync(
            context,
            HttpMethod.Get,
            "/subscriptions",
            null,
            "2020-01-01",
            cancellationToken).ConfigureAwait(false);
        var subscriptionIds = new List<string> { preferredSubscription };
        if (subscriptionsPayload.TryGetProperty("value", out var subscriptions))
        {
            subscriptionIds.AddRange(
                subscriptions.EnumerateArray()
                    .Where(item =>
                        string.Equals(
                            Text(item, "state"),
                            "Enabled",
                            StringComparison.OrdinalIgnoreCase))
                    .Select(item => Text(item, "subscriptionId"))
                    .Where(id => id is not null && !string.Equals(
                        id,
                        preferredSubscription,
                        StringComparison.OrdinalIgnoreCase))!);
        }

        var matches = new List<string>();
        foreach (var subscriptionId in subscriptionIds)
        {
            JsonElement payload;
            try
            {
                payload = await ArmAsync(
                    context,
                    HttpMethod.Get,
                    "/subscriptions/" + subscriptionId +
                    "/providers/Microsoft.Fabric/capacities",
                    null,
                    "2023-11-01",
                    cancellationToken).ConfigureAwait(false);
            }
            catch (ProvisioningRequestException exception)
                when (exception.StatusCode is HttpStatusCode.Unauthorized or HttpStatusCode.Forbidden)
            {
                continue;
            }

            if (payload.TryGetProperty("value", out var resources))
            {
                matches.AddRange(
                    resources.EnumerateArray()
                        .Where(item => string.Equals(
                            Text(item, "name"),
                            capacityName,
                            StringComparison.OrdinalIgnoreCase))
                        .Select(item => Text(item, "id"))
                        .Where(id => id is not null)!);
            }
            if (matches.Count > 0 &&
                string.Equals(
                    subscriptionId,
                    preferredSubscription,
                    StringComparison.OrdinalIgnoreCase))
            {
                break;
            }
        }

        return matches.Count switch
        {
            1 => matches[0].TrimEnd('/'),
            > 1 => throw new InvalidOperationException(
                "More than one Azure Fabric capacity matches '" + capacityName +
                "'. Use a capacity whose resource name is unique across the accessible subscriptions."),
            _ => throw new InvalidOperationException(
                "The Azure resource for Fabric capacity '" + capacityName +
                "' could not be found in the accessible subscriptions.")
        };
    }

    private async Task EnsureCapacityRoleAssignmentAsync(
        ProvisioningContext context,
        string capacityResourceId,
        string principalId,
        CancellationToken cancellationToken)
    {
        var normalized = capacityResourceId.Trim().TrimEnd('/');
        var parts = normalized.Split('/', StringSplitOptions.RemoveEmptyEntries);
        if (parts.Length != 8 ||
            !string.Equals(parts[0], "subscriptions", StringComparison.OrdinalIgnoreCase) ||
            !Guid.TryParse(parts[1], out _) ||
            !string.Equals(parts[2], "resourceGroups", StringComparison.OrdinalIgnoreCase) ||
            !string.Equals(parts[4], "providers", StringComparison.OrdinalIgnoreCase) ||
            !string.Equals(parts[5], "Microsoft.Fabric", StringComparison.OrdinalIgnoreCase) ||
            !string.Equals(parts[6], "capacities", StringComparison.OrdinalIgnoreCase))
        {
            throw new InvalidOperationException("The Azure Fabric capacity resource ID is invalid.");
        }

        var roleSeed = Encoding.UTF8.GetBytes(
            normalized.ToLowerInvariant() + "|" + principalId.ToLowerInvariant());
        var roleBytes = SHA256.HashData(roleSeed)[..16];
        roleBytes[6] = (byte)((roleBytes[6] & 0x0f) | 0x40);
        roleBytes[8] = (byte)((roleBytes[8] & 0x3f) | 0x80);
        var roleHex = Convert.ToHexString(roleBytes).ToLowerInvariant();
        var assignmentId =
            roleHex[..8] + "-" + roleHex[8..12] + "-" + roleHex[12..16] + "-" +
            roleHex[16..20] + "-" + roleHex[20..32];
        var roleDefinitionId =
            "/subscriptions/" + parts[1] +
            "/providers/Microsoft.Authorization/roleDefinitions/b24988ac-6180-42a0-ab88-20f7382dd24c";

        await ArmAsync(
            context,
            HttpMethod.Put,
            normalized +
            "/providers/Microsoft.Authorization/roleAssignments/" +
            assignmentId,
            new
            {
                properties = new
                {
                    principalId,
                    principalType = "ServicePrincipal",
                    roleDefinitionId
                }
            },
            "2022-04-01",
            cancellationToken).ConfigureAwait(false);
    }

    private async Task<JsonElement> ArmAsync(ProvisioningContext context, HttpMethod method, string path, object? body, string apiVersion, CancellationToken cancellationToken)
    {
        var token = await _tokens.AcquireForScopeAsync(context.Session, ArmScope, cancellationToken).ConfigureAwait(false);
        var separator = path.Contains('?') ? "&" : "?";
        return await SendAsync(method, ArmRoot + path + separator + "api-version=" + apiVersion, token.AccessToken, body, cancellationToken).ConfigureAwait(false);
    }

    private async Task<JsonElement> FabricAsync(ProvisioningContext context, HttpMethod method, string path, object? body, CancellationToken cancellationToken)
    {
        var token = await _tokens.AcquireForScopeAsync(context.Session, FabricScope, cancellationToken).ConfigureAwait(false);
        var url = path.StartsWith("http", StringComparison.OrdinalIgnoreCase) ? path : FabricRoot + "/" + path.TrimStart('/');
        return await SendAsync(method, url, token.AccessToken, body, cancellationToken).ConfigureAwait(false);
    }

    private async Task<List<JsonElement>> FabricCollectionAsync(ProvisioningContext context, string path, CancellationToken cancellationToken)
    {
        var items = new List<JsonElement>();
        string? next = path;
        for (var guard = 0; next is not null && guard < 100; guard++)
        {
            var payload = await FabricAsync(context, HttpMethod.Get, next, null, cancellationToken).ConfigureAwait(false);
            if (payload.TryGetProperty("value", out var value) && value.ValueKind == JsonValueKind.Array)
            {
                items.AddRange(value.EnumerateArray().Select(item => item.Clone()));
            }
            next = payload.TryGetProperty("continuationUri", out var continuation) ? continuation.GetString() : null;
        }
        return items;
    }

    private async Task<JsonElement> SendAsync(HttpMethod method, string url, string token, object? body, CancellationToken cancellationToken)
    {
        using var request = new HttpRequestMessage(method, url);
        request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
        request.Headers.Accept.Add(new MediaTypeWithQualityHeaderValue("application/json"));
        if (body is not null)
        {
            request.Content = new StringContent(JsonSerializer.Serialize(body), Encoding.UTF8, "application/json");
        }

        using var response = await _http.SendAsync(request, cancellationToken).ConfigureAwait(false);
        var text = await response.Content.ReadAsStringAsync(cancellationToken).ConfigureAwait(false);
        if (!response.IsSuccessStatusCode)
        {
            throw new ProvisioningRequestException(
                response.StatusCode,
                text,
                method.Method + " " + Redact(url) + " failed with " + (int)response.StatusCode + ".");
        }

        if (string.IsNullOrWhiteSpace(text)) return default;
        using var document = JsonDocument.Parse(text);
        return document.RootElement.Clone();
    }

    /// <summary>Removes query strings so no token or key can reach a log or an error message.</summary>
    private static string Redact(string url)
    {
        var index = url.IndexOf('?');
        return index > -1 ? url[..index] : url;
    }

    private static string? Trim(string? value) => string.IsNullOrWhiteSpace(value) ? null : value.Trim();

    private sealed record DataverseSource(
        string LakehouseId,
        string ConnectionId,
        string DeltaLakeFolder);

    private sealed record SqlEndpoint(string Id, string ConnectionString);

    private static string Require(string? value, string name) =>
        Trim(value) ?? throw new InvalidOperationException("The run is missing the required value '" + name + "'.");

    private static string? Text(JsonElement element, string property) =>
        element.TryGetProperty(property, out var value) && value.ValueKind == JsonValueKind.String ? value.GetString() : null;

    private sealed class ProvisioningRequestException(
        System.Net.HttpStatusCode statusCode,
        string responseBody,
        string message) : InvalidOperationException(message)
    {
        public System.Net.HttpStatusCode StatusCode { get; } = statusCode;
        public string ResponseBody { get; } = responseBody;
    }

public interface IArmTemplateSource
{
    JsonElement? Read();
}

/// <summary>
/// Loads the same compiled ARM template the browser uses. The file is copied from
/// deployment/azure/main.json at build time, so there is one source of truth.
/// </summary>
public sealed class FileArmTemplateSource : IArmTemplateSource
{
    private readonly string _path;
    private JsonDocument? _cached;

    public FileArmTemplateSource(string path) => _path = path;

    public JsonElement? Read()
    {
        if (_cached is not null) return _cached.RootElement;
        if (!File.Exists(_path)) return null;
        _cached = JsonDocument.Parse(File.ReadAllText(_path));
        return _cached.RootElement;
    }
}
