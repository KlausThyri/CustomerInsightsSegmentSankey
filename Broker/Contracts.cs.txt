/// <summary>
/// Wire contract shared with the Dataverse web resource. The identifiers here must
/// stay identical to the ones in webresources/segment-preview-provisioning.js and to
/// documentation/setup-center-contract.md.
/// </summary>
public static class BrokerContract
{
    public const string Version = "1.1";
    public const string HeaderName = "x-segment-preview-contract";
    public const string MessageType = "segment-preview-broker-session";

    /// <summary>Ordered step ids, identical to STEPS in the browser engine.</summary>
    public static readonly string[] Steps =
    {
        "preflight",
        "consent",
        "secret",
        "fabric-discovery",
        "fabric-notebook",
        "azure-infra",
        "fabric-permissions",
        "fabric-connection-permissions",
        "azure-app",
        "dataverse-config",
        "verify"
    };

    /// <summary>Steps the browser owns; the service must report them as skipped.</summary>
    public static readonly string[] BrowserOwnedSteps = { "preflight", "consent", "secret", "dataverse-config", "verify" };

    /// <summary>Accepts any contract whose major version matches.</summary>
    public static bool IsCompatible(string? requested)
    {
        if (string.IsNullOrWhiteSpace(requested)) return false;
        var theirs = requested.Split('.', 2)[0];
        var ours = Version.Split('.', 2)[0];
        return string.Equals(theirs.Trim(), ours, StringComparison.Ordinal);
    }
}

/// <summary>
/// The published API package the browser resolved from the solution payload.
/// The service never invents one: it deploys exactly the URL and digest the
/// caller pinned, and refuses to deploy at all when they are missing, because a
/// package-less deployment strips the code an existing Web App runs.
/// </summary>
public sealed class ApiPackage
{
    [JsonPropertyName("url")] public string? Url { get; set; }
    [JsonPropertyName("sha256")] public string? Sha256 { get; set; }
    [JsonPropertyName("blobName")] public string? BlobName { get; set; }
    [JsonPropertyName("version")] public string? Version { get; set; }

    /// <summary>True only when the package can be fetched and verified.</summary>
    [JsonIgnore]
    public bool IsUsable =>
        !string.IsNullOrWhiteSpace(Url) &&
        Uri.TryCreate(Url!.Trim(), UriKind.Absolute, out var uri) &&
        string.Equals(uri.Scheme, Uri.UriSchemeHttps, StringComparison.OrdinalIgnoreCase) &&
        IsSha256(Sha256);

    /// <summary>Blob name the template writes, digest bound so it is immutable.</summary>
    [JsonIgnore]
    public string ResolvedBlobName =>
        string.IsNullOrWhiteSpace(BlobName)
            ? "api-" + (string.IsNullOrWhiteSpace(Version) ? "0" : Version!.Trim()) + "-" +
              (Sha256 ?? string.Empty).Trim().ToLowerInvariant()[..Math.Min(16, (Sha256 ?? string.Empty).Trim().Length)] + ".zip"
            : BlobName!.Trim();

    public static bool IsSha256(string? value)
    {
        var text = (value ?? string.Empty).Trim();
        if (text.Length != 64)
        {
            return false;
        }

        foreach (var character in text)
        {
            var lower = char.ToLowerInvariant(character);
            if (!((lower >= '0' && lower <= '9') || (lower >= 'a' && lower <= 'f')))
            {
                return false;
            }
        }

        return true;
    }
}

public sealed class ErrorEnvelope
{
    [JsonPropertyName("error")] public ErrorBody Error { get; set; } = new();

    public static ErrorEnvelope Create(string code, string message, string? target = null, IEnumerable<string>? manual = null) =>
        new()
        {
            Error = new ErrorBody
            {
                Code = code,
                Message = message,
                Target = target,
                Manual = manual?.ToList() ?? new List<string>()
            }
        };
}

public sealed class ErrorBody
{
    [JsonPropertyName("code")] public string Code { get; set; } = string.Empty;
    [JsonPropertyName("message")] public string Message { get; set; } = string.Empty;
    [JsonPropertyName("target")] public string? Target { get; set; }
    [JsonPropertyName("manual")] public List<string> Manual { get; set; } = new();
}
